  const collectOutboundAliases = () => {
    const aliases = new Map()
    const visit = (value) => {
      if (!value) return
      if (Array.isArray(value)) {
        value.forEach(visit)
        return
      }
      if (typeof value !== 'object') return
      if (typeof value.id === 'string' && typeof value.tag === 'string') aliases.set(value.id, value.tag)
      Object.values(value).forEach(visit)
    }
    try {
      visit(Plugins.useProfilesStore().currentProfile?.outbounds)
    } catch (err) {}
    return aliases
  }

  const collectRouteOutbounds = () => {

    const aliases = collectOutboundAliases()
    const add = (value) => {
      if (typeof value !== 'string' || !value) return
      found.add(value)
      if (aliases.has(value)) found.add(aliases.get(value))
    }
    const visit = (value) => {
      if (!value) return
      if (Array.isArray(value)) {
        value.forEach(visit)
        return
      }
      if (typeof value !== 'object') return
      for (const [key, child] of Object.entries(value)) {
        if (key === 'outbound' || key === 'final') add(child)
        else visit(child)
      }
    }
    try {
      const route = Plugins.useProfilesStore().currentProfile?.route
      if (route?.final) add(route.final)
      visit(route)
    } catch (err) {}
    return [...found]

  // 「共享组」：被**其它组**当作子项引用的组。这类组是"子资源"，不是策略入口：
  //   ① 切换它会同时改变所有引用它的服务的出口 ⇒ 越权、破坏 per-service 隔离；
  //   ② 会被引用它的每一个 Manager 重复接管 ⇒ 多份工作量干同一件事。
  // 实测：🚀 节点选择 被 12 个场景组引用，🎈 自动选择 被 12 个组引用。
  const buildSharedGroupSet = () => {
    const shared = new Set()
    const all = kernelApi.proxies || {}
    for (const proxy of Object.values(all)) {
      if (!Array.isArray(proxy?.all)) continue
      for (const child of proxy.all) {
        if (Array.isArray(all[child]?.all)) shared.add(child)
      }
    }
    return shared
  }

  }

  // 结构性「不可接管」集合：只依赖拓扑形状，不依赖某一轮数据的具体取值（因此可以安全地用于过滤存档）。
  //  ① 被别的组当子项引用 ⇒ 共享子资源（改它 = 改所有引用它的服务，且会被多个 Manager 重复接管）
  //  ② route.final 目标 ⇒ 全局兜底出口
  //  ③ 宿主保留根选择器（GLOBAL）
  const structuralBlockSet = () => {
    const blocked = buildSharedGroupSet()
    for (const name of collectRouteFinals()) blocked.add(name)
    for (const name of RESERVED_GROUPS) blocked.add(name)
    return blocked
  }

  // T01c：**生效**的拦截集 = 结构性拦截 − 用户强制接管名单。
  // 强制接管是显式的用户选择，因此它必须同时作用于「存档过滤」与「建 Manager」这两道闸，
  // 否则会出现「面板能勾上、但一启动又被剔除」的自相矛盾。
  const effectiveBlockSet = () => {
    const blocked = structuralBlockSet()
    const released = []
    for (const name of forceManagedGroups.value) {
      if (blocked.delete(name)) released.push(name)
    }
    return { blocked, released }
  }

  // 注意：这里刻意**不**包含 hasLeafCandidatePool/isActionableGroup 这类「看当前数据才能判」的条件 ——
  // 那些条件在订阅临时加载失败时会为假，把它们用于「过滤并回写存档」会永久删掉用户的组。
  // 运行期的可行动性交给建 Manager 时判断（不落盘）。
  const isStructurallyManageable = (name) => Boolean(name) && !effectiveBlockSet().blocked.has(name)

  // Mode A 是否可用：目标必须是「可手动切换的 selector 组」且确实带叶子。
  // urltest 组不支持手改 now（switchTo 会拒绝非 selector），把它当区域子选择器只会白耗探测预算。
  const canUseModeA = (routeTarget) => {
    if (!routeTarget) return false
    const proxy = kernelApi.proxies?.[routeTarget]
    if (!proxy || !Array.isArray(proxy.all) || !proxy.all.length) return false
    if (!isSelectorGroup(routeTarget)) return false
    return proxy.all.some((id) => isLeafProxy(id))
  }

  // 该组是否「至少有一条路能产出候选」：有直接叶子（Mode B），或 now 指向可用的区域子选择器（Mode A）。
  // 两者皆无 ⇒ 建出来的 Manager 每轮只能报「没有可检测的节点池」，纯噪音（如 GLOBAL）。
  const isActionableGroup = (name) => {
    if (hasDirectLeafPool(name)) return true
    return canUseModeA(kernelApi.proxies?.[name]?.now || '')
  }

  const detectManagedGroups = (fallbackGroups = []) => {
    const groups = Object.keys(kernelApi.proxies || {}).filter((name) => isGroupProxy(name))
    const shared = buildSharedGroupSet()
    const finals = collectRouteFinals()
    const policyLike = groups.filter(
      (name) =>
        hasLeafCandidatePool(name) &&
        !shared.has(name) &&      // 子资源（如 🚀 节点选择 / 🎈 自动选择）不是策略入口
        !finals.has(name) &&      // route.final 兜底出口，不越权
        isActionableGroup(name)   // 至少有一条路能产出候选
    )
    const routeOutbounds = collectRouteOutbounds().filter((name) => policyLike.includes(name))
    const fallback = fallbackGroups.filter((name) => policyLike.includes(name))
    // T01c：强制接管的组无论拓扑上像不像「策略入口」，都要能被探测到 ——
    // 面板「自定义配置」页的可选列表正是本函数的返回值，不补进来的话用户根本看不到它们。
    const forced = forceManagedGroups.value.filter((name) => isGroupProxy(name))
    if (routeOutbounds.length) return [...new Set([...routeOutbounds, ...forced])]
    if (fallback.length) return [...new Set([...fallback, ...forced])]
    return [...new Set([...policyLike, ...forced])].sort((a, b) => a.localeCompare(b, 'zh-Hans-CN'))
  }

  const serviceKey = (label, index) => `${String(label || 'url').toLowerCase().replace(/[^a-z0-9]+/g, '-') || 'url'}-${index}`

  const labelFromUrl = (url) => {
    try {
      return new URL(url).hostname.replace(/^www\./, '').split('.')[0] || 'URL'
    } catch (err) {
      return 'URL'
    }
  }

  const servicesToText = (services = []) =>
    services.map((service) => `${service.label || labelFromUrl(service.url)}|${service.url}|${service.weight || 2}`).join('\n')

  const parseServicesText = (text) =>
    String(text || '')
      .split(/\r?\n/)
      .map((line, index) => {
        const raw = line.trim()
        if (!raw) return null
        const parts = raw.split('|').map((item) => item.trim())
        const url = parts.length >= 2 ? parts[1] : parts[0]
        if (!/^https?:\/\//i.test(url)) return null
        const label = parts.length >= 2 ? parts[0] : labelFromUrl(url)
        const weight = Number(parts[2] || 2)
        return { key: serviceKey(label, index), label, url, weight: Number.isFinite(weight) && weight > 0 ? weight : 2 }
      })
      .filter(Boolean)

  const defaultGroupConfig = (name, enabled = true) => {
    const profile = matchServiceProfile(name)
    return { name, enabled, servicesText: servicesToText(profile.services) }
  }

  const normalizeGroupConfig = (item, fallbackEnabled = true) => {
    const name = String(item?.name || '').trim()
    if (!name) return null
    const fallback = defaultGroupConfig(name, fallbackEnabled)
    return { name, enabled: item.enabled !== false, servicesText: String(item.servicesText || fallback.servicesText) }
  }

  const readStorageJson = (key, fallback) => {
    try {
      const raw = localStorage.getItem(key)
      return raw ? JSON.parse(raw) : fallback
    } catch (err) {
      return fallback
    }
  }

  const loadSchedulerConfig = (fallbackGroups = []) => {
    const currentEnabled = readStorageJson(GROUP_ENABLED_KEY, {}) || {}
    const oldEnabled = STORAGE_ID === LEGACY_STORAGE_ID ? {} : readStorageJson(LEGACY_GROUP_ENABLED_KEY, {}) || {}
    const legacyEnabled = { ...oldEnabled, ...currentEnabled }
    const saved = readStorageJson(CUSTOM_CONFIG_KEY, {})
    const oldSaved = STORAGE_ID === LEGACY_STORAGE_ID ? {} : readStorageJson(LEGACY_CUSTOM_CONFIG_KEY, {})
    const source = Array.isArray(saved.groups) ? saved : oldSaved
    const savedGroups = Array.isArray(source.groups)
      ? source.groups.map((item) => normalizeGroupConfig(item, legacyEnabled[item?.name] !== false)).filter(Boolean)
      : []

    const detected = detectManagedGroups(fallbackGroups)
    const savedMap = new Map(savedGroups.map((item) => [item.name, item]))

    // T01：用户配置优先。旧实现是 `detected.length ? detected...` —— 探测结果恒覆盖用户列表，
    // 于是用户在面板上删掉的组会被每一轮塞回来（面板"删不掉"的根因）。
    // 现在：只要面板存过组列表，就以它为准；探测结果只作**首次默认**。
    const requested = savedGroups.length ? savedGroups.map((item) => item.name) : detected
    const kernelLoaded = Object.keys(kernelApi.proxies || {}).length > 0

    // T01b：存档里的「组列表」本身也必须过结构合法性检查。
    // 为什么必须做：上面那句「用户配置优先」会把存档内容**原样放行**，于是旧版本写下的脏项
    // 会把探测层的收敛结果整个架空。真机实测（2026-09-29）：存档里有 16 个组，包含
    //   🚀 节点选择（被 12 个场景组引用的共享子资源）→ 13 组收敛失效、共享组被越权接管
    //   🐟 漏网之鱼（route.final 兜底出口）、GLOBAL（宿主保留根选择器）→ 空转 Manager
    // ★ 守卫：只在核内数据就绪时过滤；数据缺失时一律放行。绝不在信息不足时删用户的组。
    // T01c：生效的拦截集已扣除「强制接管」名单 ⇒ 用户放行的组不会在这里被剔掉。
    const { blocked, released } = kernelLoaded ? effectiveBlockSet() : { blocked: new Set(), released: [] }
    if (released.length) {
      console.info(`[smart-policy] 强制接管生效，以下组**不**参与结构性剔除：${released.join('、')}`)
    }
    const staleDropped = requested.filter((name) => blocked.has(name))
    const kept = staleDropped.length ? requested.filter((name) => !blocked.has(name)) : requested
    if (staleDropped.length) {
      console.warn(`[smart-policy] 存档中的以下组不是策略入口（共享子资源/兜底出口/保留组），已剔除：${staleDropped.join('、')}`)
    }

    // T01c：强制接管的组必须能「回到」列表里。
    // 为什么必须显式补：存档可能是上一版剔除后的结果（例如本机实测的 13 组，不含 🚀），
    // 而上面「用户配置优先」以存档为准 ⇒ 只改拦截集的话，用户的选择永远不生效。
    const forcedExisting = kernelLoaded
      ? forceManagedGroups.value.filter((name) => !kept.includes(name) && Boolean(kernelApi.proxies?.[name]))
      : []
    const withForced = forcedExisting.length ? [...kept, ...forcedExisting] : kept

    const base = kernelLoaded ? withForced.filter((name) => Boolean(kernelApi.proxies?.[name])) : withForced
    if (kernelLoaded && base.length < withForced.length) {
      const dropped = withForced.filter((name) => !base.includes(name))
      console.warn(`[smart-policy] 以下策略组在当前内核中不存在，已跳过：${dropped.join('、')}`)
    }

    // T05/D13：把两类剔除**分开返回**，因为它们的可持久化性完全不同：
    //   `persist` —— 只经过「结构性过滤 + 强制组并回」的列表 ⇒ **可以落盘**（拓扑不变量，不是临时数据）
    //   `groups`  —— 再滤掉「当前内核里不存在」的 ⇒ **不可落盘**：订阅 / provider 还没加载完时，
    //                此刻的"不存在"≠"用户删了它"，写回就会**永久删掉用户的策略组**
    //                （审计 D13 的原话：回写只写结构性过滤结果，"当前不存在"的组只跳过、不落盘）。
    const toConfig = (name) => savedMap.get(name) || defaultGroupConfig(name, legacyEnabled[name] !== false)
    const persistGroups = withForced.map(toConfig)
    const groups = base.map(toConfig)
    return { groups, persist: persistGroups }
  }

  const saveSchedulerConfig = (config) => {
    const groups = (config.groups || []).map((item) => normalizeGroupConfig(item)).filter(Boolean)
    const normalized = { groups }
    schedulerConfig.value = normalized
    localStorage.setItem(CUSTOM_CONFIG_KEY, JSON.stringify(normalized))
    localStorage.setItem(GROUP_ENABLED_KEY, JSON.stringify(Object.fromEntries(groups.map((item) => [item.name, item.enabled !== false]))))
    return normalized
  }

  // T03/T04：新参数的归一化段 —— **这里才是「能不能被外部改」的判据**（MEMORY 2.9）。
  // 刻意不用 `config.X || def`：那会把用户显式填的 0 当成「没填」，而 failoverCooldown = 0
  // 的语义恰好是「关闭独立冷却」。非有限数才回落到默认值，其余一律按用户值走。
  const posNum = (value, def, min = 0, int = false) => {
    if (value === undefined || value === null || value === '') return def
    const n = Number(value)
    if (!Number.isFinite(n)) return def
    const clamped = Math.max(min, n)
    return int ? Math.round(clamped) : clamped
  }

  const getOptions = (config = Plugin) => {
    const options = {
      ...DEFAULT_OPTIONS,
      managedGroups: parseList(config.ManagedGroups, DEFAULT_OPTIONS.managedGroups),
      monitoringInterval: Number(config.MonitoringInterval || DEFAULT_OPTIONS.monitoringInterval),
      requestTimeout: Number(config.RequestTimeout || DEFAULT_OPTIONS.requestTimeout),
      concurrencyLimit: Number(config.ConcurrencyLimit || DEFAULT_OPTIONS.concurrencyLimit),
      maxCheckNodesPerGroup: Number(config.MaxCheckNodesPerGroup || DEFAULT_OPTIONS.maxCheckNodesPerGroup),
      preserveTopNodes: Number(config.PreserveTopNodes || DEFAULT_OPTIONS.preserveTopNodes),
      rotateNodesPerCycle: Number(config.RotateNodesPerCycle || DEFAULT_OPTIONS.rotateNodesPerCycle),
      minSwitchInterval: Number(config.MinSwitchInterval || DEFAULT_OPTIONS.minSwitchInterval),
      hysteresisMargin: Number(config.HysteresisMargin || DEFAULT_OPTIONS.hysteresisMargin),
      // ---- T04 新增参数（M5/M6/M7 + M4 超时）----
      currentCheckInterval: posNum(config.CurrentCheckInterval, DEFAULT_OPTIONS.currentCheckInterval, 1000),
      currentFailureThreshold: posNum(config.CurrentFailureThreshold, DEFAULT_OPTIONS.currentFailureThreshold, 1, true),
      currentFailureWindow: posNum(config.CurrentFailureWindow, DEFAULT_OPTIONS.currentFailureWindow, 1000),
      maxSwitchPerHour: posNum(config.MaxSwitchPerHour, DEFAULT_OPTIONS.maxSwitchPerHour, 1, true),
      failoverCooldown: posNum(config.FailoverCooldown, DEFAULT_OPTIONS.failoverCooldown, 0),
      switchCallTimeout: posNum(config.SwitchCallTimeout, DEFAULT_OPTIONS.switchCallTimeout, 500),
      // ---- T05a 新增参数 ----
      nowReadTimeout: posNum(config.NowReadTimeout, DEFAULT_OPTIONS.nowReadTimeout, 300, true),
      // ---- T05 新增参数（全池覆盖率）----
      fullSweepMaxPoolSize: posNum(config.FullSweepMaxPoolSize, DEFAULT_OPTIONS.fullSweepMaxPoolSize, 0, true),
      poolCoverageTargetMs: posNum(config.PoolCoverageTargetMs, DEFAULT_OPTIONS.poolCoverageTargetMs, 0),
      poolCoverageMaxNodes: posNum(config.PoolCoverageMaxNodes, DEFAULT_OPTIONS.poolCoverageMaxNodes, 1, true)
    }
    // ★ T05：参数自洽校正 —— 拦住「快检静默失效」这条隐性事故。
    //   `CurrentFailureWindow` 必须 ≥ `CurrentFailureThreshold × CurrentCheckInterval`：
    //   否则每一次失败都在「下一次失败到来之前」过期 ⇒ 连败计数永远到不了阈值 ⇒
    //   定时器照跑、面板毫秒数照动，就是**永远不切换** —— 最坏的一种坏法：看起来一切正常。
    //   这里只把不自洽的值**抬高到刚好自洽**，用户调更大的值一律不动。
    //   校正结果写进 `options.__adjustments`，由面板「自定义配置」页显示（不静默）。
    options.__adjustments = []
    const minWindow = options.currentFailureThreshold * options.currentCheckInterval
    if (Number.isFinite(minWindow) && minWindow > 0 && options.currentFailureWindow < minWindow) {
      const from = Math.round(options.currentFailureWindow / 1000)
      const to = Math.round(minWindow / 1000)
      options.__adjustments.push(`快检失效窗口 ${from}s < 连败阈值 × 快检周期（${to}s）⇒ 已自动抬到 ${to}s，否则快检永远到不了阈值`)
      options.currentFailureWindow = minWindow
    }
    return options
  }

  const getServiceProfile = (policyGroup, groupConfig = null) => {
    const base = matchServiceProfile(policyGroup)
    const customServices = parseServicesText(groupConfig?.servicesText)
    return customServices.length ? { label: base.label, services: customServices } : base
  }

  // ============================ T03 / M1：运行代际 ============================
  // 为什么必须有：A 的巡检、切换、回读全是 await 链，而「用户点停止 / 改配置 / 内核停」是同步发生的。
  // 没有代际标记时，旧轮的 await 一返回就会照常写 lastSwitchAt、照常切换 —— 表现为
  // 「明明停了，日志还在动、组还在被改」。代际 + AbortController 是全部加固的地基。
  const createCancelledError = () => {
    const error = new Error('RUNTIME_CANCELLED')
    error.cancelled = true
    return error
  }

  const sleepWithSignal = (ms, signal) => {
    if (signal?.aborted) return Promise.reject(createCancelledError())
    return new Promise((resolve, reject) => {
      let timer = null
      const onAbort = () => {
        if (timer !== null) clearTimeout(timer)
        signal?.removeEventListener?.('abort', onAbort)
        reject(createCancelledError())
      }
      if (signal?.addEventListener) signal.addEventListener('abort', onAbort, { once: true })
      timer = setTimeout(() => {
        signal?.removeEventListener?.('abort', onAbort)
        resolve()
      }, Math.max(0, Number(ms) || 0))
    })
  }

  // T05：不可取消的等待。只用于「还没有运行代际」的阶段（例如等宿主出站表就绪）——
  // 那种场景里 `session.controller.signal` 还不存在/不该创建，用可取消版本会引入无主 AbortController。
  // 有代际的路径一律仍用 sleepWithSignal。
  const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, Math.max(0, Number(ms) || 0)))

  let runtimeGeneration = 0
  let activeRuntime = null

  const createRuntimeSession = () => {
    runtimeGeneration += 1
    const session = { id: runtimeGeneration, controller: new AbortController(), startedAt: Date.now() }
    activeRuntime = session
    return session
  }

  // ★ 唯一判据。注意：**不给参数默认值** —— 漏传 session 时必须为 false，
  //   不能退化成「只要当前有运行代际就算活着」（MEMORY 2.2：守卫漏传实参 = 同类静默失效）。
  const isSessionActive = (session) =>
    Boolean(session) && activeRuntime === session && !session.controller.signal.aborted

  const isSessionUsable = (session) => Boolean(session) && (session.detached === true || isSessionActive(session))

  // 手动操作（面板的「全检 / 巡检本组 / 切换节点」）用的**游离会话**。
  // 为什么不能复用受管会话：那些按钮在「未启动调度」的预览态也要能用，而预览态没有运行代际。
  // 为什么不能直接传 null：checkAll/switchTo 内部的每一个 await 之后都要查一次会话资格，
  //   传 null 会让它们在第一道守卫就退场（按钮变成静默 no-op —— 正是本项目最忌讳的失效形态）。
  // T05/D4：`allowSwitch` 决定这次游离巡检**是否允许自动切换真实出口**，默认 false。
  // 为什么默认不允许：「立即全检 / 重测此组」是**检测**动作，不是"接管路由"动作。
  // 旧版在预览态（未启动常驻调度）下点「立即全检」会真的把节点切掉 ——
  // 用户处于 isRunning=false / Plugin.status=0 的状态却看到出口被改，属越权（审计 D4）。
  // 要真切换有两条正路，都不用靠这个开关：
  //   ① 点节点行的「使用」= 手动 switchTo（`manual` 分支，本来就允许）；
  //   ② 启动常驻调度（用运行代际会话 `checkAll(session)`，不走 detached 分支）。
  const createDetachedSession = (opts = {}) => ({
    id: runtimeGeneration,
    controller: new AbortController(),
    detached: true,
    allowSwitch: opts.allowSwitch === true,
    startedAt: Date.now()
  })

  const cancelRuntimeSession = () => {
    const session = activeRuntime
    activeRuntime = null
    // 代际自增：让所有「只看 id 不比对象」的旧 token 一并作废
    runtimeGeneration += 1
    if (session) {
      try {
        session.controller.abort()
      } catch (err) {
        /* 宿主环境差异，忽略 */
      }
    }
    return session
  }

  // 故障类切换：走独立冷却（M7）。判据与 B 一致 —— 用原因文本特征识别，两处判据同源。
  const FAILOVER_REASON_RE = /失败|熔断|故障|不可用|中断|异常|超时|掉线|无响应/

  // ============================ T03 / M2：可取消的探测 ============================
  const setupRequestApi = () => {
    // 兜底端口按宿主区分：GUI.for.SingBox 默认 20123，GUI.for.Clash 默认 20113（原写死 9090，两者都不匹配）
    const isSingBox = String(Plugins.APP_TITLE || '').includes('SingBox')
    let base = isSingBox ? 'http://127.0.0.1:20123' : 'http://127.0.0.1:20113'
    let bearer = ''
    try {
      const profile = Plugins.useProfilesStore().currentProfile
      const clash = profile?.clash_api || profile?.experimental?.clash_api
      if (clash?.external_controller) {
        let addr = String(clash.external_controller).trim().replace(/^https?:\/\//i, '')
        if (addr.startsWith('0.0.0.0')) addr = addr.replace(/^0\.0\.0\.0/, '127.0.0.1')
        else if (addr.startsWith(':')) addr = `127.0.0.1${addr}`
        base = `http://${addr}`
      }
      if (clash?.secret) bearer = clash.secret
    } catch (err) {}
    return { base, bearer }
  }

  // T03/M2：探测可被运行代际取消（第 4 个参数 = session.controller.signal）。
  // T03/M9：控制面（内核 clash_api 本身）不可达 vs 节点不可用，必须分开归类 ——
  //   前者是「我们问不到」，后者是「节点答不上来」。混为一谈会在内核 API 打嗝时
  //   把所有节点一起判失败 ⇒ 触发一批假故障切换（B 的 PATCH4-M9 就是修这个）。
  const requestDelay = async (proxy, url, timeout, signal) => {
    if (signal?.aborted) throw createCancelledError()
    const { base, bearer } = setupRequestApi()
    const params = new URLSearchParams({ url, timeout: String(timeout) })
    const headers = bearer ? { Authorization: `Bearer ${bearer}` } : {}

    const controller = new AbortController()
    const onAbort = () => controller.abort()
    if (signal?.addEventListener) signal.addEventListener('abort', onAbort, { once: true })
    const timer = setTimeout(() => controller.abort(), timeout + 3000)

    try {
      const response = await fetch(`${base}/proxies/${encodeURIComponent(proxy)}/delay?${params.toString()}`, {
        headers,
        signal: controller.signal
      })
      const data = await response.json().catch(() => ({}))
      if (!response.ok) {
        const error = new Error(data.message || response.statusText || 'delay failed')
        // 有 HTTP 应答 ⇒ 控制面是通的，这是**节点**的问题（sing-box 用 503 表示拨号失败）
        error.httpStatus = response.status
        throw error
      }
      const delay = Number(data.delay)
      if (!Number.isFinite(delay) || delay <= 0) {
        const error = new Error(data.message || 'delay failed')
        error.httpStatus = response.status
        throw error
      }
      return delay
    } catch (error) {
      // 会话被取消 ⇒ 归因为「取消」，绝不能算成节点失败
      if (signal?.aborted || error?.cancelled) throw createCancelledError()
      if (!error?.httpStatus) error.controlPlane = true
      throw error
    } finally {
      clearTimeout(timer)
      signal?.removeEventListener?.('abort', onAbort)
    }
  }

  // ============================ T05a：运行期直读内核 now ============================
  // 为什么必须补这条通道：`kernelApi.proxies` 是**宿主的 Store**，刷新点全在面板侧
  // （面板挂载时一次 + 打开期间每 3s），**面板一关就冻结**。而 Manager 的
  // refreshCandidatePool() / currentNodeName() 读的正是这份冻结视图。
  //
  // 真机实录（2026-09-30 01:10，T03T04 上机验收）：
  //   内核已把 `🚀 节点选择` 的 now 切成「新加坡-优化2-GPT」（全池轮自己切成功的），
  //   插件视图仍停在已死的「香港WAP-优化」（对 primaryService 504），
  //   于是快检每 30s 去探一个死节点 ⇒ 它被全池轮标 OPEN ⇒ 快检静默早退
  //   ⇒ 面板恒显 `30s · -`，运维侧完全无从感知「第二条通道根本没在干活」。
  //
  // ⇒ 结论：**运行期必须以内核为唯一权威**。只读 `/proxies/{group}` 的 now，
  //   绝不写宿主 Store（写 Store 属越权，且会与宿主自身的刷新互相打架）。
  const readKernelGroupNow = async (group, signal, timeoutMs = 2000) => {
    if (!group) return { ok: false, reason: 'empty' }
    if (signal?.aborted) throw createCancelledError()
    const { base, bearer } = setupRequestApi()
    const headers = bearer ? { Authorization: `Bearer ${bearer}` } : {}

    const controller = new AbortController()
    const onAbort = () => controller.abort()
    if (signal?.addEventListener) signal.addEventListener('abort', onAbort, { once: true })
    const timer = setTimeout(() => controller.abort(), timeoutMs)

    try {
      const response = await fetch(`${base}/proxies/${encodeURIComponent(group)}`, {
        headers,
        signal: controller.signal
      })
      const data = await response.json().catch(() => ({}))
      if (!response.ok) {
        const error = new Error(data.message || response.statusText || 'read now failed')
        // 有 HTTP 应答 ⇒ 控制面是通的。404 = 组名不存在（记忆铁律 1.6），不是控制面故障。
        error.httpStatus = response.status
        throw error
      }
      // 组才有 `now`；叶子节点返回空串，调用方按「读不到」处理
      return { ok: true, now: typeof data.now === 'string' ? data.now : '', type: data.type || '' }
    } catch (error) {
      // 会话被取消 ⇒ 归因为「取消」，绝不能算成控制面故障
      if (signal?.aborted || error?.cancelled) throw createCancelledError()
      // 无 HTTP 应答 ⇒ 控制面不可达（与 requestDelay 的 M9 归类保持同一把尺子）
      if (!error?.httpStatus) error.controlPlane = true
      throw error
    } finally {
      clearTimeout(timer)
      signal?.removeEventListener?.('abort', onAbort)
    }
  }

  const asyncPool = async (limit, list, iterator) => {
    const results = []
    let index = 0
    const workers = Array.from({ length: Math.min(limit, list.length) }, async () => {
      while (index < list.length) {
        const i = index++
        try {
          results[i] = await iterator(list[i])
        } catch (err) {
          results[i] = err
        }
      }
    })
    await Promise.all(workers)
    return results
  }

  class NodeHealth {
    constructor(id, options) {
      this.id = id
      this.options = options
      this.service = {}
      this.ewmaLatency = null
      this.failureCount = 0
      this.penalty = 0
      this.state = 'CLOSED'
      this.nextAttempt = 0
      this.lastChecked = 0
      this.lastPenaltyDecay = Date.now()
      this.successRatio = 0
      // T03/M3：指标版本 + 逐类探测序号（旧结果作废的凭据）
      this.metricVersion = 0
      this.kindSeq = { full: 0, current: 0 }
      this.probeIdentity = ''
      // T04/M5：快检专用容器（约束 C1：不进 getScore）
      this.currentProbe = this.createCurrentProbeMetric()
    }

    applyPenaltyDecay() {
      const now = Date.now()
      const elapsed = now - this.lastPenaltyDecay
      if (elapsed > 1000 && this.penalty > 0 && this.options.penaltyDecayRate > 0) {
        this.penalty = Math.max(0, this.penalty * Math.exp(-this.options.penaltyDecayRate * elapsed))
        this.lastPenaltyDecay = now
      }
    }

    // ---- T04/M5：当前节点快检的**独立**指标容器 ----
    // 约束 C1（硬约定）：这个容器只服务「当前节点还活着吗」这一个判断，
    // 任何路径都不得把它写进 ewmaLatency / penalty / failureCount / successRatio。
    // 为什么：快检每次只打 1 个端点、样本极少，混进评分会污染比较基准（MEMORY 2.5）。
    createCurrentProbeMetric() {
      return { fastFailureCount: 0, windowStartedAt: 0, lastDelay: null, lastFailAt: 0, lastOkAt: 0 }
    }

    refreshCircuit() {
      if (this.state === 'OPEN' && Date.now() >= this.nextAttempt) this.state = 'HALF_OPEN'
    }

    resetCurrentProbe() {
      this.currentProbe = this.createCurrentProbeMetric()
    }

    // ---- T03/M3：探测 token ----
    // 探测身份签名变了（端点集合 / 超时改变）⇒ 旧结果必须作废，否则
    // 「拿 A 端点的成功样本给 B 端点记账」。
    markProbeIdentity(signature) {
      const next = String(signature || '')
      if (this.probeIdentity && this.probeIdentity !== next) {
        this.metricVersion += 1
        this.resetCurrentProbe()
      }
      this.probeIdentity = next
    }

    beginProbe(meta) {
      const kind = meta?.kind === 'current' ? 'current' : 'full'
      this.kindSeq[kind] += 1
      return {
        kind,
        kindSeq: this.kindSeq[kind],
        metricVersion: this.metricVersion,
        runtimeGeneration: meta?.generation,
        managerGeneration: meta?.managerGeneration
      }
    }

    // 判据：token 必须同时匹配「同一次探测（kindSeq）」「同一指标版本」「同一运行代际」「同一 manager 代际」。
    // ★ runtimeGeneration / managerGeneration 必须显式传入且为有限数 —— 不设默认值（MEMORY 2.2）。
    isTokenValid(token, runtimeGeneration, managerGeneration) {
      if (!token) return false
      if (!Number.isFinite(runtimeGeneration)) return false
      if (!Number.isFinite(managerGeneration)) return false
      if (token.runtimeGeneration !== runtimeGeneration) return false
      if (token.managerGeneration !== managerGeneration) return false
      if (token.metricVersion !== this.metricVersion) return false
      if (token.kindSeq !== this.kindSeq[token.kind]) return false
      return true
    }

    // 全池轮写指标的**唯一**入口：不过 token 一律丢弃（旧代际 / 旧探测的结果）
    recordProbe(token, results, services, runtimeGeneration, managerGeneration) {
      if (!this.isTokenValid(token, runtimeGeneration, managerGeneration)) return false
      this.record(results, services)
      return true
    }

    // ---- T04/M5：快检的结果只写 currentProbe ----
    recordCurrentProbeSuccess(delay) {
      const metric = this.currentProbe
      metric.fastFailureCount = 0
      metric.windowStartedAt = 0
      metric.lastFailAt = 0
      if (Number.isFinite(delay) && delay > 0) metric.lastDelay = delay
      metric.lastOkAt = Date.now()
    }

    // 失败必须在**抖动窗口内连续**累积：窗口外的旧失败不计入（B 的 PATCH2-P4）。
    // 返回值 = 窗口内连续失败次数。
    recordCurrentProbeFailure(windowMs) {
      const now = Date.now()
      const metric = this.currentProbe
      const window = Number.isFinite(windowMs) && windowMs > 0 ? windowMs : 60000
      if (!metric.windowStartedAt || now - metric.windowStartedAt > window) {
        metric.windowStartedAt = now
        metric.fastFailureCount = 0
      }
      metric.fastFailureCount += 1
      metric.lastFailAt = now
      return metric.fastFailureCount
    }

    isAvailable() {
      const now = Date.now()
      this.applyPenaltyDecay()
      if (this.state === 'OPEN') {
        if (now >= this.nextAttempt) {
          this.state = 'HALF_OPEN'
          return true
        }
        return false
      }
      return true
    }

    record(results, services) {
      const now = Date.now()
      this.applyPenaltyDecay()
      const isFirstSample = this.lastChecked === 0
      this.service = results
      this.lastChecked = now
      const totalWeight = services.reduce((sum, svc) => sum + svc.weight, 0)
      const okWeight = services.reduce((sum, svc) => sum + (results[svc.key]?.ok ? svc.weight : 0), 0)
      const okDelays = services.map((svc) => results[svc.key]?.delay).filter((v) => Number.isFinite(v) && v > 0)
      
      const currentRatio = totalWeight ? okWeight / totalWeight : 0
      // Smooth availability with previous history (75% current + 25% history) to resist single-packet drop
      this.successRatio = isFirstSample ? currentRatio : 0.75 * currentRatio + 0.25 * this.successRatio

      if (okDelays.length) {
        const avg = okDelays.reduce((sum, v) => sum + v, 0) / okDelays.length
        this.ewmaLatency = this.ewmaLatency === null ? avg : this.options.ewmaAlpha * avg + (1 - this.options.ewmaAlpha) * this.ewmaLatency
      }

      if (this.successRatio >= this.options.minSwitchSuccessRatio) {
        this.failureCount = 0
        this.state = 'CLOSED'
        this.penalty *= 0.6
      } else {
        this.failureCount += 1
        this.penalty += this.options.penaltyIncrement
        if (this.failureCount >= this.options.failureThreshold || this.state === 'HALF_OPEN') {
          this.state = 'OPEN'
          this.nextAttempt = now + this.options.circuitBreakerTimeout
        }
      }
    }

    getScore() {
      if (!this.isAvailable()) return -Infinity
      const latencyScore = this.ewmaLatency ? calcLatencyScore(this.ewmaLatency) : 0
      const availabilityScore = this.options.availabilityWeight * this.successRatio
      const halfOpenPenalty = this.state === 'HALF_OPEN' ? 200 : 0
      return availabilityScore + latencyScore - this.penalty * this.options.penaltyWeight - this.failureCount * 20 - halfOpenPenalty
    }
  }

  class SmartPolicyManager {
    constructor(policyGroup, options, onUpdate) {
      this.policyGroup = policyGroup
      this.groupConfig = options.groupConfigMap?.get(policyGroup) || defaultGroupConfig(policyGroup)
      this.profile = getServiceProfile(policyGroup, this.groupConfig)
      this.services = this.profile.services
      // T04/M5：快检只打 1 个端点（省钱、够判活），取本组权重最高的那个业务端点
      this.primaryService = [...this.services].sort((a, b) => (b.weight || 0) - (a.weight || 0))[0] || COMMON_SERVICES[0]
      this.options = options
      this.onUpdate = onUpdate
      this.routeTarget = ''
      this.candidateGroup = ''
      this.modeASuppressed = false
      this.nodes = []
      this.nodeMap = new Map()
      this.timer = null
      this.initialTimer = null
      this.checking = false
      this.switching = false
      this.enabled = this.groupConfig.enabled !== false
      this.current = null
      this.lastSwitchAt = 0
      this.lastReason = ''
      this.lastCheckedAt = 0
      this.scanCursor = 0
      this.lastPlanText = ''
      this.logs = []
      // ---- T03：运行代际与切换事务 ----
      this.session = null
      this.managerGeneration = 0
      this.switchSeq = 0
      this.activeSwitchSeq = 0
      this.activeSwitch = null
      this.pendingSwitch = null
      this.switchHistory = []
      this.lastFailoverAt = 0
      this.lastRoundCostMs = 0
      // ---- T04：当前节点快检 ----
      this.currentTimer = null
      this.currentTicking = false
      this.controlPlaneNotifiedAt = 0
      // ---- T05d：快检通道的「运行模式」与「粘性读数」----
      // probeMode：'preview' = 尚未启动常驻心跳（预览态）；'runtime' = 定时器在跑。
      //   为什么必须显式区分：预览态不起定时器 ⇒ currentTickCount 恒 0 ⇒ 面板只能显示「未出拍」。
      //   那是个**误导性读数** —— 它读起来像"通道坏了"，真相却是"你还没点启动"；
      //   而且点「立即全检」也救不了（那条路径走全池轮，不写快检字段）。
      //   真机 2026-09-30 用户报告："快检心跳 启动啦还显示未出拍"。
      this.probeMode = 'preview'
      // lastKnownDelay：最后一次探测成功拿到的延迟，**跨拍保留**。
      //   用途：候选池被清空（拓扑未就绪）时 this.current 会变 null，
      //   旧版 currentDelayText 于是立刻掉回 '-'，看起来像"刚才那轮白测了"。
      this.lastKnownDelay = null
      // ---- T05a：运行期直读内核真值 ----
      // kernelNowMap 是**本 Manager 自己的** now 视图，优先级高于宿主 Store。
      // 只增不减（组名有限），无需清理；代际切换时由 startMonitoring 清空重建。
      this.kernelNowMap = new Map()
      this.kernelNowAt = 0
      this.kernelNowFailures = 0
      // ---- T05c：快检心跳（把「看不见」变成「看得见」）----
      // 判据：只要定时器把这一拍交到 tickCurrent（不管后续是否早退）就 +1。
      // 用途：面板「快检心跳」卡片显示 `N 拍 · M 秒前`。
      //   若 N 不涨 ⇒ 通道真的死了；若 N 在涨而 currentDelayText 恒为 `-` ⇒ 拍到了但没结果（可查 skipReason）。
      //   这正是本轮排查时最缺的那个信号 —— 当时面板恒显 `-`，无法区分「没跑」与「跑了但没结果」。
      this.currentTickCount = 0
      this.lastCurrentTickAt = 0
      this.lastCurrentNote = ''
      // ---- T05：全池轮停滞自检（watchdog）----
      // 快检是**唯一每 30s 必然跑一次**的通道 ⇒ 借它做"全池轮还活着吗"的判据最省。
      // 真机 2026-09-30 出现过一段 ≈25 分钟零探测的窗口，旧版对这种「整条全池轮停摆」
      // 完全没有信号（面板只会显示上一次的旧读数）。
      this.roundWatchFrom = 0
      this.roundStaleNote = ''
      this.roundStaleNotifiedAt = 0
    }

    addLog(level, message) {
      const item = { id: `${Date.now()}-${Math.random().toString(16).slice(2)}`, time: Date.now(), level, message }
      this.logs = [item, ...this.logs].slice(0, this.options.logLimit || 120)
      console.log(`[${Plugin.name}][${this.policyGroup}]`, message)
    }

    // T03/M9：控制面（内核 clash_api）不可达时的留痕。限频 5 分钟，避免刷屏。
    noteControlPlaneDown() {
      const now = Date.now()
      if (now - this.controlPlaneNotifiedAt < 300000) return
      this.controlPlaneNotifiedAt = now
      this.addLog('warn', '无法访问内核 API，本次探测不计入节点失败统计（请检查内核运行状态与 clash_api 凭据）')
    }

    startMonitoring(session, initialDelay = 0) {
      this.stopMonitoring()
      this.session = session
      this.managerGeneration += 1
      // T05d：定时器即将就位 ⇒ 从这一刻起这张卡显示的是"真心跳"，而不是预览态占位。
      this.probeMode = 'runtime'
      // T05c：新代际重新计数心跳。旧代际的读数留着只会让人误以为「新代际也在跑」。
      this.currentTickCount = 0
      this.lastCurrentTickAt = 0
      this.lastCurrentNote = ''
      // T05：watchdog 的起点 = 本次启动时刻（首轮完成前用它可以避免误报）
      this.roundWatchFrom = Date.now()
      this.roundStaleNote = ''
      this.roundStaleNotifiedAt = 0
      this.initialTimer = setTimeout(() => {
        this.runRound()
        this.runCurrentTick()
        this.timer = setInterval(() => this.runRound(), this.options.monitoringInterval)
        // T04/M5：第二条通道 —— 只打当前节点，周期与全池轮独立
        this.currentTimer = setInterval(() => this.runCurrentTick(), this.options.currentCheckInterval)
      }, initialDelay)
    }

    // 定时器回调统一入口：取消不算异常（停机/重配置时的正常路径，不打日志）
    runRound() {
      this.checkAll(this.session).catch((error) => {
        if (error?.cancelled || !isSessionActive(this.session)) return
        lastError.value = error.message || String(error)
        this.addLog('warn', lastError.value)
        this.onUpdate()
      })
    }

    runCurrentTick() {
      this.tickCurrent(this.session).catch((error) => {
        if (error?.cancelled || !isSessionActive(this.session)) return
        this.addLog('warn', `当前节点快检异常：${error.message || String(error)}`)
      })
    }

    stopMonitoring() {
      if (this.initialTimer) clearTimeout(this.initialTimer)
      if (this.timer) clearInterval(this.timer)
      if (this.currentTimer) clearInterval(this.currentTimer)
      this.initialTimer = null
      this.timer = null
      this.currentTimer = null
      // T03：交还运行代际持有权 —— 在途的探测/切换/回读回来时 isSessionActive 恒为 false，
      // 于是它们不会写任何状态。这是「停机彻底化」（原 D2）的实现方式。
      this.session = null
      this.managerGeneration += 1
      // T05d：停机后退回预览态 —— 否则面板会继续显示一个**冻结的**心跳数字
      //   （看起来通道还在跑），与"已停止"的总体状态自相矛盾。
      this.probeMode = 'preview'
      this.switching = false
      this.activeSwitchSeq = 0
      this.activeSwitch = null
      this.pendingSwitch = null
      this.checking = false
      this.currentTicking = false
      // T05a：清空直读缓存（MEMORY 2.25「停用路径必须清空数据容器」）。
      // 停机后继续展示上一次的 now 会误导 —— 用户会以为是「当前真值」。
      // 心跳计数**不清**：它是「曾经跑过多少拍」的证据，停后仍有诊断价值（与 logs 同理）。
      this.kernelNowMap = new Map()
      this.kernelNowFailures = 0
    }

    refreshCandidatePool() {
      const policy = kernelApi.proxies[this.policyGroup]
      if (!policy) return
      // ★ T05/D12：拓扑数据未就绪时必须**什么都不做**。
      //   旧写法的 Mode A 准入条件是 `!hasDirectLeafPool(policyGroup)`，而 `hasDirectLeafPool`
      //   在数据缺失时恰好返回 false ⇒ **缺数据 = 走 Mode A**，等于把"不知道"当成"是"。
      //   （旧版靠 `if (!policy) return` 兜住了一部分，但 `policy` 存在而 `policy.all` 缺失时兜不住。）
      //   改成正向表达：只有确认拿到完整 `all` 数组才算数；否则清空候选池并留痕 ——
      //   最坏结果是"这一轮不检测"，绝不会是"这一轮乱切一个组"。
      if (!Array.isArray(policy.all)) {
        this.candidateGroup = ''
        this.nodes = []
        this.nodeMap = new Map()
        this.current = null
        this.lastReason = `内核拓扑数据未就绪（${this.policyGroup} 缺少 all 数组），本轮不检测`
        return
      }
      // T05a：routeTarget 同样**以直读值为准**。Store 的 now 在面板关闭后是冻结的，
      // 用它会算错候选组（真机：内核已是 新加坡-优化2-GPT，Store 还是 香港WAP-优化）。
      const freshRoute = this.kernelNowMap.get(this.policyGroup)
      this.routeTarget = (typeof freshRoute === 'string' && freshRoute ? freshRoute : policy.now) || ''
      const routeProxy = kernelApi.proxies[this.routeTarget]
      let candidateGroup = ''
      let candidateIds = []

      // Mode A: 2-Tier Hierarchy (Policy -> Region Sub-selector -> Leaf nodes)
      // T01/D9 守卫：仅当本组自己「没有直接叶子池」时才按 2 层结构处理。
      // 若本组本身就带叶子（扁平池，如各场景组），now 指向某个组只是「借用了共享组」——
      // 此时按 Mode A 去切那个共享组，等于越权改动所有引用它的服务的出口
      // （实测：8 个场景组的 now 都是 🚀 节点选择 ⇒ 8 个 Manager 一起去切它、各自 now 永不动）。
      // 正确行为是退回 Mode B，走本组自有的叶子池，per-service 隔离才成立。
      //
      // T01c 追加守卫：若 `now` 指向的组**本身也在受管集里**，同样不许用 Mode A 去写它。
      // 为什么：同一个组被两个 Manager 交替改写 = 把 D9 换个入口重新引入。
      // 真机实例：🐟 漏网之鱼（强制接管）的 now 就是 🚀 节点选择（同被强制接管）——
      // 若不拦，🐟 的 Manager 会拿自己的业务画像去改 🚀，与 🚀 自己的 Manager 抢同一个选择器。
      // 被拦下后该组会落到「没有可检测的节点池」，这是**预期结果**：它只管自己，不代管别人。
      const targetIsManaged = (this.options.managedGroups || []).includes(this.routeTarget)
      this.modeASuppressed = false
      if (!hasDirectLeafPool(this.policyGroup) && !targetIsManaged && canUseModeA(this.routeTarget)) {
        candidateGroup = this.routeTarget
        // T05a：routeTarget 现在可能来自直读值（Store 里未必有此键）⇒ 必须空值安全
        candidateIds = (routeProxy?.all || []).filter((id) => isLeafProxy(id))
      } else if (!hasDirectLeafPool(this.policyGroup) && targetIsManaged && canUseModeA(this.routeTarget)) {
        this.modeASuppressed = true
      }

      // Mode B: 1-Tier Flat Hierarchy (Policy -> Leaf nodes directly)
      if (!candidateIds.length && Array.isArray(policy.all)) {
        candidateGroup = this.policyGroup
        candidateIds = policy.all.filter((id) => isLeafProxy(id))
      }

      this.candidateGroup = candidateGroup
      const old = this.nodeMap
      this.nodes = candidateIds.map((id) => old.get(id) || new NodeHealth(id, this.options))
      this.nodeMap = new Map(this.nodes.map((node) => [node.id, node]))
      // T03/M3：探测端点集合是节点指标的「身份」之一 —— 它变了，旧结果就不能再记账
      // （否则「拿 A 端点的成功样本给 B 端点用」，MEMORY 2.10 的同类失效）。
      const identity = `${this.candidateGroup}|${this.services.map((svc) => svc.url).join(',')}|${this.options.requestTimeout}`
      this.nodes.forEach((node) => node.markProbeIdentity(identity))
      const nowNode = this.currentNodeName()
      this.current = this.nodes.find((node) => node.id === nowNode) || this.current
    }

    async checkAll(session = this.session) {
      // T03：代际守卫。停机/重配置后，旧的 await 链回到这里必须直接退场。
      if (!isSessionUsable(session)) return
      if (this.checking) return
      if (!this.enabled) {
        if (this.lastReason !== '已关闭智能调度') {
          this.lastReason = '已关闭智能调度'
          this.addLog('muted', this.lastReason)
        }
        this.onUpdate()
        return
      }
      this.checking = true
      const startedAt = Date.now()
      try {
        // T05a：与 tickCurrent 同一纪律 —— 先取内核真值，再算候选池。
        // 全池轮同样依赖 policyGroup.now 定候选组，用陈旧值会「整轮都在探错的池」。
        await this.syncKernelNow(session)
        if (!isSessionUsable(session)) return
        this.refreshCandidatePool()
        if (!isSessionUsable(session)) return
        if (!this.nodes.length) {
          this.lastReason = this.modeASuppressed
            ? `策略组 ${this.policyGroup} 无自有节点池；其 now（${this.routeTarget}）也在受管集里，已让位给它自己的调度器（不重复接管）`
            : `策略组 ${this.policyGroup} 当前选择 ${this.routeTarget || '-'}，没有可检测的节点池`
          this.addLog('warn', this.lastReason)
          return
        }
        const plan = this.buildCheckPlan()
        this.lastPlanText = `本轮检测 ${plan.length}/${this.nodes.length} 个节点`
        this.addLog('info', `${this.lastPlanText}，候选池 ${this.candidateGroup || '-'}，当前 ${this.currentNodeName() || '-'}`)
        await asyncPool(this.options.concurrencyLimit, plan, (node) => this.checkNode(node, session))
        // ★ 本轮结束前再查一次代际：停机发生在探测期间时，下面每一句都是「写入」，
        //   一律不许执行（这是「停了还在动」的最后一道闸）。
        if (!isSessionUsable(session)) return
        this.lastCheckedAt = Date.now()
        const ready = this.nodes.filter((node) => node.successRatio >= this.options.minSwitchSuccessRatio).length
        this.addLog(ready ? 'success' : 'warn', `检测完成，可用 ${ready}/${this.nodes.length}`)
        // ★ T05/D4：游离开的「手动巡检」默认**只检测、不切换**。
        //   判据只看会话形状，不看 isRunning —— 因为"预览态"和"运行中点了重测"用的是同一种会话，
        //   两种情形下用户的意图都是"看一眼结果"，而不是"替我改路由"。
        //   真切换的入口：启动常驻调度（走运行代际会话）或点节点行的「使用」。
        const dryRun = Boolean(session && session.detached) && session.allowSwitch !== true
        await this.evaluateSwitch(session, { dryRun })
      } finally {
        this.lastRoundCostMs = Date.now() - startedAt
        // 单轮超过间隔 ⇒ 实际节奏比配置值慢。不清不楚地变慢是「调度器看着没毛病却反应迟钝」的常见来源。
        if (this.lastRoundCostMs > this.options.monitoringInterval && isSessionUsable(session)) {
          this.addLog('info', `单轮耗时 ${Math.round(this.lastRoundCostMs / 1000)}s，已超过检测间隔 ${Math.round(this.options.monitoringInterval / 1000)}s，实际节奏会变慢`)
        }
        this.checking = false
        if (isSessionUsable(session)) this.onUpdate()
      }
    }

    buildCheckPlan() {
      const currentName = this.currentNodeName()
      const current = this.nodes.find((node) => node.id === currentName)
      
      const fresh = this.nodes.filter((node) => node.id !== currentName && node.lastChecked === 0)
      const checked = this.nodes.filter((node) => node.id !== currentName && node.lastChecked > 0).sort((a, b) => b.getScore() - a.getScore())
      const stale = [...checked].sort((a, b) => a.lastChecked - b.lastChecked)

      // Fast Warmup check: If more than 30% of nodes in pool are fresh, scale up budget temporarily
      const isWarmup = fresh.length > 3 && fresh.length / this.nodes.length > 0.3
      const baseBudget = Math.max(1, Number(this.options.maxCheckNodesPerGroup) || 4)
      const warmupBudget = isWarmup ? Math.min(10, Math.max(baseBudget, Math.ceil(this.nodes.length * 0.35))) : 0
      // T05：三者取最大值。sweepBudget 负责"全池覆盖"，是消灭 25 倍差距的那一项。
      const maxBudget = Math.max(baseBudget, warmupBudget, this.poolSweepBudget())

      const picked = []
      const seen = new Set()

      const add = (node) => {
        if (!node || seen.has(node.id) || picked.length >= maxBudget) return false
        seen.add(node.id)
        picked.push(node)
        return true
      }

      if (current) add(current)

      // In warmup phase, take up to 4 fresh nodes to build leaderboards quickly
      const freshQuota = isWarmup ? Math.min(4, fresh.length) : (fresh.length ? 1 : 0)
      fresh.slice(0, freshQuota).forEach(add)

      const rotateLimit = Math.max(1, Number(this.options.rotateNodesPerCycle) || 1)
      this.pickRotatingNodes(stale, rotateLimit).forEach(add)

      const preserveTop = Math.max(1, Number(this.options.preserveTopNodes) || 2)
      checked.slice(0, preserveTop).forEach(add)

      fresh.forEach(add)
      stale.forEach(add)

      return picked
    }

    // T05：单轮巡检预算（全池覆盖率）。
    // 旧版固定 `maxCheckNodesPerGroup`(默认 4) ⇒ 27 池要 ≈101 分钟才轮到一遍，
    // 而 B 是"全池 27 个 / 3 分钟" ⇒ 差约 25 倍。M5 快检只打当前节点，不解决这一条。
    // 规则（两条互补，都是"结构性"修正，不是调参）：
    //   ① 池规模 ≤ fullSweepMaxPoolSize ⇒ 本轮**全量**（27 池 15 分钟一轮 ⇒ 差距 25 倍 → 5 倍）
    //   ② 更大的池 ⇒ 按 `poolCoverageTargetMs` 反推单轮预算：
    //      needed = ceil(poolSize × interval / target)，再以 poolCoverageMaxNodes 封顶，
    //      防止 96~130 成员的池一次性把探测预算打满。
    poolSweepBudget() {
      const size = this.nodes.length
      if (!size) return 1
      const sweepCap = Number(this.options.fullSweepMaxPoolSize) || 0
      if (sweepCap > 0 && size <= sweepCap) return size
      const target = Number(this.options.poolCoverageTargetMs) || 0
      if (target <= 0) return 1
      const interval = Math.max(1, Number(this.options.monitoringInterval) || 1)
      const needed = Math.ceil((size * interval) / target)
      const capped = Math.min(Math.max(1, Number(this.options.poolCoverageMaxNodes) || 1), Math.max(1, needed))
      return Math.max(1, capped)
    }

    pickRotatingNodes(list, limit) {
      if (!list.length || limit <= 0) return []
      const picked = []
      const start = this.scanCursor % list.length
      for (let index = 0; index < Math.min(limit, list.length); index += 1) picked.push(list[(start + index) % list.length])
      this.scanCursor = (start + picked.length) % list.length
      return picked
    }

    async checkNode(node, session = this.session) {
      if (!isSessionUsable(session)) return
      if (!node.isAvailable()) return
      // T03/M3：先领 token，结果回来时用它证明「这次探测仍是当前有效的这一次」
      const token = node.beginProbe({ kind: 'full', generation: session.id, managerGeneration: this.managerGeneration })
      const results = {}
      let controlPlaneFailures = 0
      await asyncPool(this.options.serviceConcurrency, this.services, async (service) => {
        try {
          const delay = await requestDelay(node.id, service.url, this.options.requestTimeout, session.controller.signal)
          results[service.key] = { ok: true, delay }
        } catch (error) {
          if (error?.cancelled) {
            results[service.key] = { ok: false, cancelled: true, error: 'cancelled' }
            return
          }
          if (error?.controlPlane) controlPlaneFailures += 1
          results[service.key] = { ok: false, error: error.message || String(error) }
        }
      })
      // 会话已失效 ⇒ 本轮结果一律丢弃，绝不写指标
      if (!isSessionUsable(session)) return
      // T03/M9：全部端点都栽在「问不到内核」上 ⇒ 这是控制面故障，不是节点故障，不计节点账
      if (controlPlaneFailures === this.services.length) {
        this.noteControlPlaneDown()
        return
      }
      node.recordProbe(token, results, this.services, session.id, this.managerGeneration)
    }

    // ---- T04/M6：每小时切换上限（1 小时滑动窗口）----
    // 为什么必须有：M5 让故障发现进入 60s 级，若无额度约束，10 分钟就能把一小时的名额烧光，
    // 之后整小时「静默罢工」（B 的 PATCH3-P2 真实事故）。丢弃过期时间戳的同时顺带回收内存。
    canSwitch() {
      const now = Date.now()
      this.switchHistory = this.switchHistory.filter((time) => now - time < 3600000)
      return this.switchHistory.length < this.options.maxSwitchPerHour
    }

    // ---- T05：全池轮停滞自检 ----
    // 判据：距「上一轮全池巡检**完成**」的时刻已经超过 2 × 巡检间隔。
    //   - 首轮完成前用 `roundWatchFrom`（启动时刻）当基准，避免刚启动就误报；
    //   - 只有 `enabled` 的组会走到这里（tickCurrent 在未启用时已早退），不会对被关掉的组误报；
    //   - 停表后 currentTimer 也被清掉 ⇒ 停机状态不会产生假告警。
    // 留痕限频 10 分钟，避免 16 组 × 每 30s 把日志刷爆。
    checkRoundWatchdog() {
      const interval = Math.max(60000, Number(this.options.monitoringInterval) || 0)
      const since = this.lastCheckedAt || this.roundWatchFrom
      if (!since) return
      const idle = Date.now() - since
      if (idle <= interval * 2) {
        this.roundStaleNote = ''
        return
      }
      this.roundStaleNote = `全池轮已停滞 ${Math.round(idle / 60000)} 分钟（间隔 ${Math.round(interval / 60000)} 分钟）`
      if (Date.now() - this.roundStaleNotifiedAt < 600000) return
      this.roundStaleNotifiedAt = Date.now()
      this.addLog('warn', `${this.roundStaleNote}：快检仍在运行，但全池择优结果已停止更新（请检查调度是否仍处于运行态）`)
    }

    // ---- T04/M5：当前节点快检（第二条探测通道）----
    // 与全池轮的关系：工作集不重叠（全池轮按候选池抽样，这里只打当前节点），
    // 因此**不互斥**（互斥只会白丢检测轮次）。与在途切换只做必要隔离：快检不得覆盖切换结果。
    async tickCurrent(session = this.session, opts = {}) {
      // T05d：preview = 预览态单次检测。与 runtime 的差别**只有两点**：不计心跳、禁止故障切换。
      //   其余（读内核真值 → 算候选池 → 探测当前节点 → 写指标）完全同源，避免出现两套口径（D7 同类问题）。
      const preview = opts.preview === true
      if (!isSessionUsable(session)) return
      if (this.currentTicking) {
        // T05d/P0-1：这条早退以前是**完全静默**的 —— 心跳不涨、无日志、面板无痕迹，
        //   用户只能看到"数字不涨"而拿不到任何原因。现在留痕。
        this.lastCurrentNote = '上一拍尚未结束（本拍跳过，避免并发探测同一节点）'
        return
      }
      if (!this.enabled) {
        this.lastCurrentNote = '组未启用（本组调度已关闭）'
        return
      }
      this.currentTicking = true
      // T05c：心跳在此计数 ——「拍数增长」= 通道真的在执行任务。
      // 反过来说：计数不涨只有两种可能 —— 通道被停掉，或上一拍卡住（currentTicking 恒为 true）。
      // 这两种在旧版都是**不可见**的（面板只会恒显 `-`），正是本轮 T05 要消灭的盲区。
      // T05d：预览态不计心跳 —— 那个数字的语义是"定时器拍数"，而预览态没有定时器。
      if (!preview) {
        this.currentTickCount += 1
        this.lastCurrentTickAt = Date.now()
      }
      this.lastCurrentNote = ''
      // T05：全池轮停滞自检。放在这里是因为快检是唯一"每 30s 必然跑一次"的通道。
      // 预览态既没有定时器也没有全池轮，不做停滞判定（否则一进面板就误报"全池轮已停滞"）。
      if (!preview) this.checkRoundWatchdog()
      try {
        // T05a：★ 顺序不能反 —— 必须先取内核真值，再算候选池。
        //   refreshCandidatePool 用 policyGroup.now 定 routeTarget，用 Store 的旧值会算错候选组；
        //   而 currentNodeName 用 candidateGroup.now 取当前节点，陈旧值会导致「对着过期节点探测」。
        await this.syncKernelNow(session)
        if (!isSessionUsable(session)) {
          this.lastCurrentNote = '会话已失效（停机或重配置），本拍中止'
          return
        }
        this.refreshCandidatePool()
        if (!isSessionUsable(session)) {
          this.lastCurrentNote = '会话已失效（停机或重配置），本拍中止'
          return
        }
        const current = this.nodeMap.get(this.currentNodeName()) || null
        this.current = current
        if (!current) {
          // T05d/P0-1：「拓扑未就绪」与「真的找不到该节点」是两件事，读数必须能区分开。
          this.lastCurrentNote = /未就绪/.test(this.lastReason)
            ? this.lastReason
            : `候选池里找不到当前节点（now=${this.currentNodeName() || '读不到'}）`
          return
        }
        current.refreshCircuit()

        // ---- T05b：熔断期不再静默早退 ----
        // 旧行为 `if (current.state === 'OPEN') return` 让「最需要观察恢复的时刻」彻底闭嘴：
        // 不出探测、不出日志、不写 lastDelay ⇒ 面板恒显 `-`，运维侧零感知
        // （真机 2026-09-30 实录：正是这一行 + Store 陈旧，一起制造了「快检看起来根本没跑」的假象）。
        // 而快检的职责本就是回答「当前节点还活着吗」—— 这里当然包括「从熔断中恢复了吗」。
        // 安全性：失败路径仍走 failoverFromCurrent，那里有 M7 冷却(180s) + M6 额度双闸门，
        //   不会因为熔断期照探而反复切换（下面还有一层冷却窗口内的静默短路）。
        const wasCircuitOpen = current.state !== 'CLOSED'
        if (wasCircuitOpen) this.lastCurrentNote = `节点熔断中（${current.state}），照探等待恢复`

        const token = current.beginProbe({ kind: 'current', generation: session.id, managerGeneration: this.managerGeneration })
        let result = null
        try {
          const delay = await requestDelay(current.id, this.primaryService.url, this.options.requestTimeout, session.controller.signal)
          result = { ok: true, delay }
        } catch (error) {
          if (error?.cancelled) {
            this.lastCurrentNote = '本拍已取消（停机或重配置）'
            return
          }
          result = { ok: false, error: error.message || String(error), controlPlane: Boolean(error?.controlPlane) }
        }

        // 每个 await 之后重新校验资格（T03/M4 的同一套纪律）
        // T05d/P0-1：下面三条早退以前都是**静默**的（心跳在涨、延迟恒 `-`、状态却显示"正常"）
        //   —— 这正是用户看到的「三个自相矛盾的读数」。现在每条都留痕。
        if (!isSessionUsable(session)) {
          this.lastCurrentNote = '会话已失效（停机或重配置），本拍结果作废'
          return
        }
        if (!this.nodeMap.has(current.id)) {
          this.lastCurrentNote = `节点 ${current.id} 已不在候选池（拓扑变化），本拍结果作废`
          return
        }
        if (!current.isTokenValid(token, session.id, this.managerGeneration)) {
          this.lastCurrentNote = `结果已过期（token 失效），本拍作废：${current.id}`
          this.addLog('info', `快检结果丢弃（token 已过期）：${current.id}`)
          return
        }
        if (result.controlPlane) {
          this.noteControlPlaneDown()
          this.lastCurrentNote = '控制面不可达（本次不计入节点失败）'
          return
        }
        if (result.ok) {
          current.recordCurrentProbeSuccess(result.delay)
          // T05d：粘性读数 —— 供候选池被清空时面板仍能显示"最近一次测到多少"。
          this.lastKnownDelay = result.delay
          this.lastCurrentNote = ''
          // T05b：熔断期照探成功 ⇒ 立刻解除熔断。这正是「照探」换来的收益：
          // 旧版在熔断期内完全静止，节点恢复了也要等最长 circuitBreakerTimeout 才被重新发现。
          if (wasCircuitOpen) {
            current.state = 'CLOSED'
            current.failureCount = 0
            current.nextAttempt = 0
            this.addLog('success', `当前节点快检恢复正常：${current.id}（${Math.round(result.delay)}ms）—— 已解除熔断`)
          }
          return
        }

        const consecutive = current.recordCurrentProbeFailure(this.options.currentFailureWindow)
        // ★ 只在「刚越过阈值」那一拍出声，之后静默；恢复时由上面的 success 分支出声。
        //   为什么不能每拍都报：16 组 × 每 30s ⇒ 每 2 秒一条，会把真正的异常淹掉。
        //   「快检有没有在跑」的可观测性改由面板「快检心跳」卡片承担（T05c）。
        if (consecutive === this.options.currentFailureThreshold) {
          this.addLog('warn', `当前节点快检连续失败 ${consecutive}/${this.options.currentFailureThreshold}：${current.id}（${result.error}）`)
        }
        if (consecutive < this.options.currentFailureThreshold) {
          // T05d/P0-1：★ 这条以前是 `return`（什么都不写）—— 于是「探测失败但未越阈值」这一格
          //   面板上呈现为「延迟 = `-` + 状态 = 正常」，是三个读数里最误导的一个。
          //   日志仍按上面的理由限频，但**面板状态必须每拍都写**。
          this.lastCurrentNote = `探测失败 ${consecutive}/${this.options.currentFailureThreshold}（${result.error}）`
          return
        }

        // T05d：预览态到此为止 —— 只报告"该切了"，绝不真的切（D4 的同一原则）。
        if (preview) {
          this.lastCurrentNote = `预览态：当前节点已连续失败 ${consecutive} 次（${result.error}），启动调度后将紧急切换`
          return
        }

        // T05b：熔断期照探的失败 ⇒ 仍处于故障冷却窗口内就静默返回。
        // 首次熔断那一拍已经触发过一次 failover（日志已出），冷却期内每 30s 重复报一句
        // 「冷却中」只会刷屏 —— M6/M7 的本意是「少动作」，不是「多说话」。
        // T05d/P0-1：日志依旧静默，但面板状态要写（让卡片能回答"为什么坏了却没换"）。
        const cooldown = this.options.failoverCooldown
        if (wasCircuitOpen && cooldown > 0 && Date.now() - this.lastFailoverAt < cooldown) {
          const left = Math.ceil((cooldown - (Date.now() - this.lastFailoverAt)) / 1000)
          this.lastCurrentNote = `当前节点仍故障；故障切换冷却中（剩余 ${left}s）`
          return
        }

        // ★ 顺序不能反：先置 OPEN 再调 failoverFromCurrent —— 后者开头会把 CLOSED 节点直接拒掉
        current.state = 'OPEN'
        current.nextAttempt = Date.now() + this.options.circuitBreakerTimeout
        const switched = await this.failoverFromCurrent(session, `当前节点快检 ${Math.round(this.options.currentFailureWindow / 1000)}s 内连续失败 ${consecutive} 次`)
        if (!switched && isSessionUsable(session)) {
          // 没切走就别把监控也关掉：留着 OPEN 会让后续每拍快检全部空转，
          // 也发现不了节点已经恢复（最长 circuitBreakerTimeout 的完全静止）。
          // ★ T05b 起这条更关键：照探能否生效，取决于这里把节点置成 HALF_OPEN。
          current.state = 'HALF_OPEN'
          current.nextAttempt = Date.now() + Math.max(1000, this.options.currentCheckInterval)
          // T05d/P0-1：没切走的原因（无候选 / 冷却中 / 额度用尽）已在 lastReason 里，
          //   转写到快检状态上，让这张卡能直接回答"为什么当前节点坏了却没换"。
          this.lastCurrentNote = `当前节点故障但未切换：${this.lastReason}`
        } else if (switched) {
          this.lastCurrentNote = '当前节点故障，已故障切换到备用节点（等待下一拍快检给出新读数）'
        }
      } finally {
        this.currentTicking = false
        if (isSessionUsable(session)) this.onUpdate()
      }
    }

    // ---- T04/M7：故障切换（独立冷却 + 额度）----
    async failoverFromCurrent(session, reason) {
      if (!isSessionUsable(session)) return false
      const current = this.current
      if (!current || current.state === 'CLOSED') return false
      const candidates = this.nodes
        .filter((node) => node.id !== current.id && node.getScore() > -Infinity && node.successRatio >= this.options.minSwitchSuccessRatio)
        .sort((a, b) => b.getScore() - a.getScore())
      if (!candidates.length) {
        this.lastReason = `当前节点故障，但候选池里没有可用节点，暂不切换`
        this.addLog('warn', `${this.lastReason}：${reason}`)
        return false
      }
      const now = Date.now()
      const cooldown = this.options.failoverCooldown
      if (cooldown > 0 && now - this.lastFailoverAt < cooldown) {
        const left = Math.ceil((cooldown - (now - this.lastFailoverAt)) / 1000)
        this.lastReason = `故障切换冷却中（剩余 ${left}s），本次跳过`
        this.addLog('info', `${this.lastReason}：${reason}`)
        return false
      }
      if (!this.canSwitch()) {
        // 额度用尽是「静默罢工」的源头，必须留痕（宁可刷屏也不能让它无声无息）
        this.lastReason = `每小时切换额度已用尽（${this.switchHistory.length}/${this.options.maxSwitchPerHour}）`
        this.addLog('warn', `${this.lastReason}，故障切换被跳过：${reason}`)
        return false
      }
      return await this.switchTo(candidates[0], reason, session)
    }

    async evaluateSwitch(session = this.session, opts = {}) {
      const dryRun = opts.dryRun === true
      if (!isSessionUsable(session)) return
      const candidates = this.nodes.filter((node) => node.getScore() > -Infinity && node.successRatio >= this.options.minSwitchSuccessRatio)
      if (!candidates.length) {
        this.lastReason = `候选节点均未达到最低${this.profile.label}可达性要求，暂不切换`
        this.addLog('warn', this.lastReason)
        return
      }
      const best = candidates.sort((a, b) => b.getScore() - a.getScore())[0]
      const currentName = this.currentNodeName()
      const current = this.nodeMap.get(currentName) || null
      this.current = current || this.current

      // ★ T05/D4：预览态（游离开 + 未放行切换）走「只检测」分支。
      //   判据与真实决策**完全同源**（同一 candidates、同一 getScore 口径），只是把"执行"换成"记录建议" ——
      //   这样面板上看到的结论与启动后真的会做的事一致，不会出现"两套口径"（D7 的同类问题）。
      if (dryRun) {
        if (!current) {
          this.lastReason = `预览态：当前节点 ${currentName || '读不到'} 不在候选池；启动调度后将切到 ${best.id}`
          this.addLog('info', this.lastReason)
          return
        }
        if (best.id === current.id) {
          this.lastReason = `预览态：当前节点仍是最优 ${best.id}（分数 ${formatScore(best.getScore())}）`
          this.addLog('success', this.lastReason)
          return
        }
        const dead = !current.isAvailable() || current.successRatio < this.options.minSwitchSuccessRatio
        if (dead) {
          this.lastReason = `预览态：当前节点故障（可用率 ${Math.round(current.successRatio * 100)}%）；启动调度后将紧急切换到 ${best.id}`
          this.addLog('warn', this.lastReason)
          return
        }
        const gain = Math.round(best.getScore() - current.getScore())
        if (gain >= this.options.hysteresisMargin) {
          this.lastReason = `预览态：发现更优节点 ${best.id}（+${gain}）；启动调度后会在冷却允许时切换`
          this.addLog('info', this.lastReason)
        } else {
          this.lastReason = `预览态：最佳节点 ${best.id} 未超过滞后阈值（+${gain}/${this.options.hysteresisMargin}），保持当前选择`
          this.addLog('info', this.lastReason)
        }
        return
      }

      if (!current) {
        if (!this.canSwitch()) {
          this.lastReason = `每小时切换额度已用尽（${this.switchHistory.length}/${this.options.maxSwitchPerHour}）`
          this.addLog('warn', `${this.lastReason}，首次接管被跳过`)
          return
        }
        await this.switchTo(best, '当前节点不在候选池，切换到最佳节点', session)
        return
      }

      const now = Date.now()
      const currentScore = current.getScore()
      const bestScore = best.getScore()

      if (best.id === current.id) {
        this.lastReason = `当前节点仍是最优: ${best.id}`
        this.addLog('success', `${this.lastReason}，分数 ${formatScore(bestScore)}`)
        return
      }

      const isCurrentDead = !current.isAvailable() || current.successRatio < this.options.minSwitchSuccessRatio

      if (isCurrentDead) {
        await this.failoverFromCurrent(session, `当前节点故障(可用率 ${Math.round(current.successRatio * 100)}%)，触发紧急自愈切换`)
        return
      }

      // Shared candidate pool anti-oscillation protection
      const lastPoolSwitch = candidatePoolSwitchMap.get(this.candidateGroup) || 0
      const timeSinceLastPoolSwitch = now - Math.max(this.lastSwitchAt, lastPoolSwitch)

      if (timeSinceLastPoolSwitch < this.options.minSwitchInterval) {
        const waitSeconds = Math.ceil((this.options.minSwitchInterval - timeSinceLastPoolSwitch) / 1000)
        this.lastReason = `发现更优节点 ${best.id}，但所属候选池处于切换冷却期`
        this.addLog('info', `${this.lastReason}，剩余约 ${waitSeconds}s`)
        return
      }

      // T04/M6：普通（性能更优）路径同样受每小时额度约束 —— 与故障路径同一把尺子，避免判据漂移
      if (!this.canSwitch()) {
        this.lastReason = `每小时切换额度已用尽（${this.switchHistory.length}/${this.options.maxSwitchPerHour}）`
        this.addLog('warn', `${this.lastReason}，切换被跳过：目标 ${best.id}`)
        return
      }

      if (bestScore >= currentScore + this.options.hysteresisMargin) {
        await this.switchTo(best, `综合分数提升 ${Math.round(bestScore - currentScore)}`, session)
      } else {
        const delta = Math.round(bestScore - currentScore)
        this.lastReason = `最佳节点 ${best.id} 未超过滞后阈值，保持当前选择`
        this.addLog('info', `${this.lastReason}，提升 ${delta} / 阈值 ${this.options.hysteresisMargin}`)
      }
    }

    // 带超时的 Promise 包装。只用来兜住「宿主 API 挂死」——
    // 宿主 handleUseProxy 不可取消是它的固有属性，超时后只能作废事务，不能真的中断它。
    callWithTimeout(promise, ms) {
      let timer = null
      const timeout = new Promise((resolve, reject) => {
        timer = setTimeout(() => {
          const error = new Error('SWITCH_CALL_TIMEOUT')
          error.switchTimeout = true
          reject(error)
        }, ms)
      })
      return Promise.race([promise, timeout]).finally(() => {
        if (timer !== null) clearTimeout(timer)
      })
    }

    // T03/M4：事务资格判据 —— 代际 + 序号，两者缺一不可。
    // 只比值不够（stop 之后 restart 可能拿到同一个 managerGeneration 之外的旧 tx），
    // 只比代际也不够（同一代际内的两次切换会互相覆盖回读结果）。
    isSwitchValid(tx, session) {
      if (!isSessionUsable(session)) return false
      if (!tx || tx.seq !== this.activeSwitchSeq) return false
      if (tx.managerGeneration !== this.managerGeneration) return false
      return true
    }

    // 宿主 Plugins.handleUseProxy 的真实契约（宿主产物 index-*.js 中的 bl 函数）：
    //   ① 严格判断 `type === 'Selector'`，否则静默 return；
    //   ② 目标已是当前节点时静默 return，什么都不做；
    //   ③ 是 async 且无返回值 —— 调用方不 await 就完全不知道成败。
    // 因此必须 await + 回读内核状态，并且只在「确实切过去了」之后才写切换冷却。
    //
    // T03/M4：升级为四阶段事务 prepare → execute → verify → commit。
    // 已被占用时不丢弃请求，改入 pendingSwitch 排队（旧版直接 return false 丢失一次故障自救机会）。
    async switchTo(node, reason, session = this.session) {
      // ★ 手动切换（面板点节点行）不隶属任何运行代际：预览态下 manager.session 为 null，
      //   若一律按 isSessionActive 拦，手动切换会静默失效（回归）。因此显式区分「无代际的手动切换」。
      const manual = !session
      const sessionValid = () => (manual ? true : isSessionUsable(session))
      const txValid = () => (manual ? tx.seq === this.activeSwitchSeq : this.isSwitchValid(tx, session))

      if (!sessionValid()) return false
      if (!node) return false

      const previous = this.currentNodeName()
      // 目标已是当前节点：宿主会静默忽略，这里也直接短路。关键是**不能写切换冷却**，
      // 否则用户误点「当前节点」那一行，就会把自动调度冻结一整个 minSwitchInterval。
      if (previous === node.id) {
        this.current = node
        this.lastReason = `${reason}: 已是当前节点 ${node.id}，无需切换`
        this.addLog('info', this.lastReason)
        return true
      }

      const isFailover = FAILOVER_REASON_RE.test(String(reason || ''))
      this.switchSeq += 1
      const tx = {
        id: `${this.managerGeneration}-${this.switchSeq}`,
        seq: this.switchSeq,
        from: previous,
        target: node.id,
        generation: session ? session.id : 0,
        managerGeneration: this.managerGeneration,
        phase: 'prepare',
        reason,
        isFailover
      }

      if (this.switching) {
        // 同目标去重 / 故障类插队 / 非故障不覆盖已排队的故障
        const queued = this.pendingSwitch
        const shouldReplace = !queued || isFailover || queued.target === tx.target || !queued.isFailover
        if (shouldReplace) this.pendingSwitch = { ...tx, phase: 'queued' }
        this.lastReason = `已有切换请求进行中，本次排队（${previous || '-'} -> ${node.id}）`
        return false
      }

      this.activeSwitchSeq = tx.seq
      this.activeSwitch = tx
      this.switching = true
      try {
        // ---------------- prepare ----------------
        if (!txValid()) return false
        const groupProxy = kernelApi.proxies[this.candidateGroup]
        const nodeProxy = kernelApi.proxies[node.id]
        if (!groupProxy || !nodeProxy) {
          this.lastReason = '未找到候选组或目标节点代理对象，无法切换'
          this.addLog('warn', this.lastReason)
          return false
        }
        const groupType = String(groupProxy.type || '').toLowerCase()
        if (groupType && groupType !== 'selector') {
          this.lastReason = `分组 [${this.candidateGroup}] 类型为 ${groupProxy.type}，非 Selector，无法手动指定节点`
          this.addLog('warn', this.lastReason)
          return false
        }

        // ---------------- execute ----------------
        tx.phase = 'execute'
        try {
          // 包一层 then：宿主实现若同步抛错，也要走同一条失败路径（Promise.race 不接同步异常）
          await this.callWithTimeout(
            Promise.resolve().then(() => Plugins.handleUseProxy(groupProxy, nodeProxy)),
            this.options.switchCallTimeout
          )
        } catch (error) {
          if (!txValid()) return false
          if (error?.switchTimeout) {
            // 超时 ⇒ 本事务作废，且必须丢弃排队目标：否则锁一释放，排队的切换会立刻再撞同一个挂死调用
            this.pendingSwitch = null
            this.lastReason = `调用宿主切换超过 ${this.options.switchCallTimeout}ms 未返回，已放弃本次事务`
            this.addLog('warn', `${this.lastReason}：${previous || '-'} -> ${node.id}`)
            return false
          }
          this.lastReason = `切换 ${previous || '-'} -> ${node.id} 调用失败: ${error.message || String(error)}`
          this.addLog('warn', this.lastReason)
          return false
        }

        // ---------------- verify（回读内核，不信任宿主返回值） ----------------
        const applied = await this.waitForApplied(node.id, session)
        if (!txValid()) return false
        if (!applied) {
          this.lastReason = `切换 ${previous || '-'} -> ${node.id} 未生效（宿主未确认，可能是分组类型非 Selector 或代理尚未就绪）`
          this.addLog('warn', this.lastReason)
          return false
        }
        tx.phase = 'verify'
        if (!txValid()) return false
        if (this.currentNodeName() !== node.id) return false

        // ---------------- commit ----------------
        tx.phase = 'commit'
        const switchedAt = Date.now()
        this.current = node
        // 新节点接任，快检的失败计数不得继承旧节点
        node.resetCurrentProbe()
        this.lastSwitchAt = switchedAt
        if (isFailover) this.lastFailoverAt = switchedAt
        this.switchHistory.push(switchedAt)
        // T05/D6：顺手回收过期键（旧版只写不删）
        prunePoolSwitchMap(switchedAt)
        candidatePoolSwitchMap.set(this.candidateGroup, switchedAt)
        this.lastReason = `${reason}: ${this.candidateGroup} -> ${node.id}`
        this.addLog('switch', `${reason}: ${previous || '-'} -> ${node.id}`)
        return true
      } finally {
        // 只有「自己仍是当前唯一有效事务」时才释放锁并消费队列；
        // 旧事务晚回来时既不解锁（新事务正持有），也不消费队列。
        if (this.activeSwitchSeq === tx.seq) {
          this.switching = false
          this.activeSwitch = null
          this.drainPendingSwitch(session)
        }
      }
    }

    // 消费排队的切换事务。★ 必须在 finally、switching 已复位之后调用，并二次校验代际与目标。
    drainPendingSwitch(session) {
      const next = this.pendingSwitch
      if (!next) return
      this.pendingSwitch = null
      if (!isSessionUsable(session)) return
      if (next.generation !== session.id) return
      if (next.managerGeneration !== this.managerGeneration) return
      if (this.currentNodeName() === next.target) return
      const node = this.nodeMap.get(next.target)
      if (!node) return
      this.switchTo(node, next.reason, session).catch(() => {})
    }

    // ---- T05a：运行期直读内核 now ----
    // 单组直读。除「会话被取消」外**不抛异常**，返回 `{ ok, controlPlane? }`。
    // 控制面故障（M9）只留痕、绝不停摆：内核 API 打嗝一次不该让快检整轮消失。
    async refreshKernelGroupNow(group, session) {
      if (!group) return { ok: false }
      // session 为空 ⇒ 手动 / 游离路径（不隶属任何运行代际）：允许直读，只是不做资格校验
      const hasSession = Boolean(session)
      if (hasSession && !isSessionUsable(session)) return { ok: false }
      try {
        const result = await readKernelGroupNow(group, session?.controller?.signal, this.options.nowReadTimeout)
        if (hasSession && !isSessionUsable(session)) return { ok: false }
        if (!result.ok) return { ok: false }
        this.kernelNowMap.set(group, result.now || '')
        this.kernelNowAt = Date.now()
        this.kernelNowFailures = 0
        return { ok: true }
      } catch (error) {
        if (error?.cancelled) return { ok: false, cancelled: true }
        this.kernelNowFailures += 1
        if (error?.controlPlane) {
          this.noteControlPlaneDown()
          return { ok: false, controlPlane: true }
        }
        // 有 HTTP 应答（如组名 404）⇒ 这是**局部**问题，不影响其它组继续读
        this.addLog('info', `读取内核 now 失败（${group}）：${error.message || error}`)
        return { ok: false }
      }
    }

    // 刷新本轮要用的全部 now。范围 = 本组 + 候选组 + 当前路由目标（去重，最多 3 个请求）。
    // 为什么三者都要：refreshCandidatePool 用 policyGroup.now 定 routeTarget，
    // 而 currentNodeName 用 candidateGroup.now 取当前节点 —— 任一陈旧都会导致「对着过期节点工作」。
    // ★ 必须筛掉「不是组」的名字：Mode B 下 routeTarget 是一个**节点名**，
    //   读它会得到没有 now 字段的叶子对象（白费一次请求，还会往 map 里塞垃圾键）。
    async syncKernelNow(session) {
      if (!isSessionUsable(session)) return false
      const isGroupName = (name) => Array.isArray(kernelApi.proxies?.[name]?.all)
      const groups = [...new Set([this.policyGroup, this.candidateGroup, this.routeTarget].filter((name) => name && isGroupName(name)))]
      let anyOk = false
      for (const group of groups) {
        if (!isSessionUsable(session)) return anyOk
        const result = await this.refreshKernelGroupNow(group, session)
        if (result.ok) anyOk = true
        // 控制面不可达 ⇒ 后面几个请求注定也失败，别白等（最坏能省 4s）
        else if (result.controlPlane) break
      }
      return anyOk
    }

    // 等待内核回读确认切换落地：宿主 refreshProviderProxies 是异步的，PUT 返回那一刻不代表 `now` 已更新。
    // T03/M2：等待本身可被 session 取消（否则停止后仍会空转 2.4s）。
    // T05a：★ 回读必须走**直读**。用宿主 Store 回读等于「拿陈旧视图确认落地」——
    //   面板关闭时 Store 根本不刷新，会一直判不落地（真机表现为白等 2.4s 后放弃切换确认）。
    async waitForApplied(nodeId, session, tries = 12, gapMs = 200) {
      for (let index = 0; index < tries; index += 1) {
        // session 为 null ⇒ 手动切换，不隶属任何运行代际，只需轮询到落地
        if (session && !isSessionUsable(session)) return false
        const readResult = await this.refreshKernelGroupNow(this.candidateGroup, session)
        if (this.currentNodeName() === nodeId) return true
        // 控制面不可达 ⇒ 后面几轮注定也读不到，提前放弃（否则 12×2s 会把 2.4s 的等待拖成 24s）
        if (readResult && readResult.controlPlane) return false
        try {
          await sleepWithSignal(gapMs, session?.controller?.signal)
        } catch (error) {
          return false
        }
      }
      return false
    }

    // 当前节点名。★ T05a 起**以内核直读值为第一权威**，宿主 Store 只作回落。
    // 原因见 readKernelGroupNow 的注释：Store 的刷新点全在面板侧，面板一关就冻结。
    currentNodeName() {
      const group = this.candidateGroup
      if (!group) return ''
      const fresh = this.kernelNowMap.get(group)
      if (typeof fresh === 'string' && fresh) return fresh
      return kernelApi.proxies[group]?.now || ''
    }

    syncExternalState(silent = false) {
      const previousRoute = this.routeTarget
      const previousPool = this.candidateGroup
      const previousCurrent = this.currentNodeName()
      this.refreshCandidatePool()
      const nextCurrent = this.currentNodeName()
      const changed = previousRoute !== this.routeTarget || previousPool !== this.candidateGroup || previousCurrent !== nextCurrent
      if (changed && !silent) {
        this.lastReason = `检测到外部选择变化：${previousRoute || '-'} -> ${this.routeTarget || '-'}`
        this.lastPlanText = ''
        this.addLog('info', `${this.lastReason}，当前节点 ${nextCurrent || '-'}`)
      }
      return changed
    }

    // T05d：当前节点延迟的**取值链**（把三个来源合成一个读数）。
    //   ① 快检成功值（currentProbe.lastDelay）—— 最准，但只在 runtime 通道产出；
    //   ② 全池轮里当前节点的首服务延迟（node.service[primaryService.key].delay）——
    //      预览态点「立即全检」走的就是这条；旧版不认它，于是"点了半天死活不出数字"；
    //   ③ 粘性上一拍值（lastKnownDelay）—— 候选池被清空（拓扑未就绪）时兜底，
    //      避免"刚测出来的读数"凭空消失、看起来像白测了一轮。
    //   返回 `-` 只在"从未测到过"时出现 —— 那才是真的没有读数。
    resolveCurrentDelayText() {
      const fast = this.current?.currentProbe?.lastDelay
      if (Number.isFinite(fast)) return `${Math.round(fast)}ms`
      const svc = this.current?.service?.[this.primaryService?.key]
      if (svc && svc.ok && Number.isFinite(svc.delay)) return `${Math.round(svc.delay)}ms`
      if (Number.isFinite(this.lastKnownDelay)) return `~${Math.round(this.lastKnownDelay)}ms`
      return '-'
    }

    // T05d：延迟读数的**来源标签** —— 让面板能说清"这个数字是哪条通道测出来的"。
    //   没有它就等于把三个来源压成一个不可区分的数（本项目最忌讳的失效形态）。
    currentDelaySource() {
      if (Number.isFinite(this.current?.currentProbe?.lastDelay)) return '快检通道'
      const svc = this.current?.service?.[this.primaryService?.key]
      if (svc && svc.ok && Number.isFinite(svc.delay)) return '上一轮全池检测'
      if (Number.isFinite(this.lastKnownDelay)) return '更早的一拍（读数可能已过期）'
      return '暂无'
    }

    // T05d/P0-1：状态必须由**单一来源**推导。
    //   旧版是「note 为空 ⇒ 显示'正常'」，而"探测失败但未达阈值"那条路径恰好不写 note
    //   ⇒ 面板出现「延迟 `-` + 状态 正常」的自相矛盾组合（用户 2026-09-30 报告）。
    //   现在三段式：有 note 用 note；无 note 且拿到延迟 ⇒ 正常(带毫秒)；无 note 也无延迟 ⇒ 明确说"未产出"。
    currentStatus() {
      if (this.lastCurrentNote) return this.lastCurrentNote
      if (this.probeMode !== 'runtime') {
        return `预览态：快检通道未启动（点「启动调度」后每 ${Math.round(this.options.currentCheckInterval / 1000)}s 一拍）`
      }
      const delay = this.current?.currentProbe?.lastDelay
      if (Number.isFinite(delay)) return `正常 · ${Math.round(delay)}ms`
      return '本拍未产出结果（无延迟读数，请查看日志）'
    }

    toView() {
      const currentName = this.currentNodeName()
      const serviceColumns = this.services.slice(0, 5)
      const rows = [...this.nodes].sort((a, b) => b.getScore() - a.getScore()).map((node) => {
        const services = serviceColumns.map((svc) => {
          const item = node.service[svc.key]
          const ok = item?.ok
          const delay = ok ? Math.round(item.delay) : null
          let cls = 'd-none'
          if (!item) cls = 'd-none'
          else if (!ok) cls = 'd-fail'
          else if (delay < 200) cls = 'd-fast'
          else if (delay < 500) cls = 'd-mid'
          else cls = 'd-slow'
          return { key: svc.key, text: serviceText(node, svc.key), delay: ok ? delay : 999999, cls }
        })
        const score = node.getScore()
        return {
          id: node.id,
          selected: node.id === currentName,
          score: formatScore(score),
          scoreNum: score === -Infinity ? -999999 : score,
          successRatio: `${Math.round(node.successRatio * 100)}%`,
          successRatioNum: node.successRatio,
          services,
          ewma: node.ewmaLatency ? `${Math.round(node.ewmaLatency)}ms` : '-',
          ewmaNum: node.ewmaLatency || 999999,
          failureCount: node.failureCount,
          stateText: stateText(node.state),
          stateCls: node.state === 'CLOSED' ? 's-normal' : node.state === 'OPEN' ? 's-open' : 's-half'
        }
      })
      const bestRow = rows[0]
      return {
        policyGroup: this.policyGroup,
        routeTarget: this.routeTarget || '-',
        candidateGroup: this.candidateGroup || '-',
        current: currentName || '-',
        profileLabel: this.profile.label,
        serviceColumns,
        enabled: this.enabled,
        bestNode: bestRow?.id || '-',
        bestRatio: bestRow?.successRatio || '-',
        readyNodes: this.nodes.filter((node) => node.successRatio >= this.options.minSwitchSuccessRatio).length,
        totalNodes: this.nodes.length,
        lastReason: this.lastReason,
        lastPlanText: this.lastPlanText,
        // T04：快检通道的可观测面（验收时要能一眼看出「快检在跑、当前节点延迟多少」）
        // T05d：改为**取值链** —— 旧版只认 currentProbe（快检专有），导致两个误读：
        //   ① 候选池被清空时 this.current 变 null ⇒ 立刻掉回 `-`（像是"刚才那轮白测了"）；
        //   ② 预览态点「立即全检」走的是全池轮（写 node.service，不写 currentProbe）
        //      ⇒ 用户点了半天，这张卡死活不出数字（真机 2026-09-30 用户报告的直接原因）。
        currentDelayText: this.resolveCurrentDelayText(),
        currentDelaySource: this.currentDelaySource(),
        fastFailCount: this.current?.currentProbe?.fastFailureCount || 0,
        currentCheckText: `${Math.round(this.options.currentCheckInterval / 1000)}s`,
        switchesLastHour: this.switchHistory.filter((time) => Date.now() - time < 3600000).length,
        maxSwitchPerHour: this.options.maxSwitchPerHour,
        // T05a/T05c：直读与心跳的可观测面。
        // 这几个字段的存在意义 = 「把看不见变成看得见」：旧版只能靠 `-` 猜，现在能直接读出
        //   「跑了多少拍 / 上一拍什么时候 / 上一拍为什么没产出结果 / 内核对不上号几次」。
        currentTickCount: this.currentTickCount,
        lastCurrentTickAt: this.lastCurrentTickAt,
        // T05d/P0-2：预览态必须说"还没启动"，而不是"未出拍"——
        //   后者读起来是"通道坏了"，而真相是"你还没点启动"；且旧版点「立即全检」也救不了它，
        //   于是用户点得越多越确信坏了（真机 2026-09-30 用户报告）。
        probePreview: this.probeMode !== 'runtime',
        currentTickText: this.probeMode !== 'runtime'
          ? '未启动（预览态）'
          : (this.currentTickCount
              ? `${this.currentTickCount} 拍 · ${Math.max(0, Math.round((Date.now() - this.lastCurrentTickAt) / 1000))}s前`
              : '未出拍'),
        lastCurrentNote: this.lastCurrentNote || '',
        // T05d/P0-1：状态的**唯一**导出点（三段式推导，见 currentStatus()）。
        currentStatusText: this.currentStatus(),
        // T05：全池轮停滞状态（watchdog）。空串 = 正常。
        roundStaleNote: this.roundStaleNote || '',
        // T05/D7：决策阈值必须能从视图里读到 —— 面板的「只看健康」过滤要用**同一个数**，
        // 否则一旦阈值可调，两处立刻分叉（勾了"只看健康"却混进不合格节点）。
        minRatio: this.options.minSwitchSuccessRatio,
        // T05：单轮巡检预算（全池覆盖率的可观测面）
        checkBudget: this.poolSweepBudget(),
        kernelNowAt: this.kernelNowAt,
        kernelNowFailures: this.kernelNowFailures,
        kernelNowSource: this.kernelNowMap.size ? '直读内核' : '宿主Store',
        services: summarizeServices(this.nodes, this.services),
        logs: this.logs.map((item) => ({ ...item, timeText: formatLogTime(item.time) })),
        rows
      }
    }
  }

  const serviceText = (node, key) => {
    const item = node.service[key]
    if (!item) return '-'
    return item.ok ? `${Math.round(item.delay)}ms` : '失败'
  }

  const summarizeServices = (nodes, services) =>
    services.map((svc) => {
      const ok = nodes.filter((node) => node.service[svc.key]?.ok).length
      return { key: svc.key, label: svc.label, ok: ok > 0, text: nodes.length ? `${ok}/${nodes.length}` : '-' }
    })

  const stateText = (state) => (state === 'CLOSED' ? '正常' : state === 'OPEN' ? '熔断' : state === 'HALF_OPEN' ? '试探' : state || '-')
  const formatLogTime = (time) => {
    const d = new Date(time)
    return [d.getHours(), d.getMinutes(), d.getSeconds()].map((v) => String(v).padStart(2, '0')).join(':')
  }
  const formatScore = (score) => (score === -Infinity ? '-' : Math.round(score))

  // T03：重入保护。start() 是异步的，而它可能被 4 个钩子同时唤起
  // （onReady / onCoreStarted / onEnabled / onReload）。两个并发的 start 会各自 createRuntimeSession，
  // 后者的会话成为 activeRuntime，前者建出的那批 manager 就永远拿不到「有效会话」——
  // 表现为「一半的组永远停在等待启动」。这类竞态不会抛错，只会静默失效。
  let starting = false

  const start = async (config = Plugin) => {
    if (starting) return 0
    starting = true
    try {
      return await startRuntime(config)
    } finally {
      starting = false
    }
  }

  // T05/D11：等宿主出站表就绪。
  // 为什么需要：`kernelApi.running` 置位的时刻，`proxies` 往往**还是空的**（它是异步加载的）。
  // 旧版只判 `running` ⇒ 开机自启（`on::ready`）与内核重启自恢复（`on::core::started`）这两条路
  // 大概率直接 throw「未找到可接管的策略组」 ⇒ 用户看到的现象是「自动启动没生效，手动点一下又好了」。
  // 这里先**主动请宿主刷新一次**（宿主只在「代理页挂载 / 切换节点后」刷新，没有定时轮询），
  // 再轮询等待；等不到也不算致命 —— 由调用方决定怎么报错。
  const waitForKernelProxies = async (timeoutMs = 10000, stepMs = 300) => {
    const ready = () => Object.keys(kernelApi.proxies || {}).length > 0
    if (ready()) return true
    try {
      if (typeof kernelApi.refreshProviderProxies === 'function') await kernelApi.refreshProviderProxies()
    } catch (err) {}
    if (ready()) return true
    const deadline = Date.now() + Math.max(0, Number(timeoutMs) || 0)
    while (Date.now() < deadline) {
      await sleep(stepMs)
      if (ready()) return true
    }
    return ready()
  }

  // T05/D8：把「构建运行实例」与「提交（停旧 + 起新）」彻底拆开。
  // 旧版 `startRuntime` 第一行就是 `stop(false, true)` —— 于是配置有问题 / 内核还没就绪时，
  // **旧 runtime 已经被杀掉且没有回滚**，用户看到的现象是「改一下配置，调度就没了」（审计 D8）。
  // 现在：先纯计算地构建出全部 Manager，全部成功后才动现状。
  const buildManagerPlan = (config = Plugin) => {
    const options = getOptions(config)
    const loaded = loadSchedulerConfig(options.managedGroups)
    // D5：去重。同组双 Manager = 双倍探测 + 冷却键互相覆盖 + 面板双行。
    // （D6 的"冷却被覆盖"其实源自这里 —— 去重后同一个 candidateGroup 天然只有一个写入者。）
    const seen = new Set()
    const persistGroups = loaded.persist.filter((item) => {
      if (seen.has(item.name)) return false
      seen.add(item.name)
      return true
    })
    const byName = new Map(persistGroups.map((item) => [item.name, item]))
    // 建 Manager 用 loaded.groups（已滤掉"当前内核里不存在"的），
    // 落盘用 persistGroups（保留它们 ⇒ 订阅/provider 没加载完时不会被永久删除）。
    const manageable = loaded.groups
      .map((item) => item.name)
      .filter((name, index, list) => list.indexOf(name) === index)
      .filter((name) => isStructurallyManageable(name) && Boolean(kernelApi.proxies[name]))
    options.managedGroups = manageable
    options.groupConfigMap = new Map(manageable.map((name) => [name, byName.get(name) || defaultGroupConfig(name)]))
    const created = manageable.map((group) => {
      const manager = new SmartPolicyManager(group, options, notifyUpdate)
      manager.refreshCandidatePool()
      return manager
    })
    return { options, created, persistGroups }
  }

  const startRuntime = async (config = Plugin) => {
    lastError.value = ''
    if (!kernelApi.running) throw new Error('核心未运行，无法启动分流策略智能调度')
    const ready = await waitForKernelProxies()
    if (!ready) {
      throw new Error('内核出站表尚未就绪（proxies 为空）。请稍等片刻重试，或先在 App 的「代理」页停留一次')
    }
    // ★ D8：先构建、后提交。构建过程中任何 throw 都不会动到正在跑的 runtime。
    const plan = buildManagerPlan(config)
    if (!plan.created.length) {
      const names = plan.persistGroups.map((item) => item.name).join(', ')
      throw new Error(`未找到可接管的策略组: ${names || '（配置为空）'}`)
    }
    if (!plan.created.some((manager) => manager.nodes.length)) {
      // 不致命（例如配置里只剩 🐟/GLOBAL 这类设计上就空转的组），但要留痕
      console.warn('[smart-policy] 已找出受管组，但候选池全为空 —— 内核拓扑可能尚未就绪')
    }
    // ---------- 到这里才提交（前面任何一步失败都不会影响正在跑的调度）----------
    stop(false, true)
    // T05/D13：落盘只写「结构性过滤 + 去重」结果，**不写**"当前不存在"的组
    saveSchedulerConfig({ groups: plan.persistGroups })
    // T03/M1：代际只在「确实要开始跑」的时刻产生 —— 上面提前 throw 的路径不会留下无主的运行代际
    const session = createRuntimeSession()
    managers.value = plan.created
    managers.value.forEach((manager, index) => manager.startMonitoring(session, index * plan.options.groupStartDelay))
    isRunning.value = true
    Plugin.status = 1
    notifyUpdate()
    return 1
  }

  const stop = (setStatus = true, clearManagers = false) => {
    // T03：先作废运行代际并 abort —— 在途的 fetch / 回读轮询立刻失去「写状态」的资格。
    // 注意顺序：必须先 abort 再停表，否则旧轮的 await 可能在停表与 abort 之间落地。
    cancelRuntimeSession()
    managers.value.forEach((manager) => manager.stopMonitoring())
    if (clearManagers) managers.value = []
    isRunning.value = false
    if (setStatus) Plugin.status = 0
    notifyUpdate()
  }

  const ensurePreviewManagers = () => {
    if (managers.value.length || !kernelApi.running) return
    // T05：预览态**不再落盘**。旧版这里调 `saveSchedulerConfig(...)`，
    // 于是「只是打开一次面板」就会改写持久配置（审计 P2「打开面板即落盘」）。
    // 预览只是"看一看"，不该有持久副作用。
    const plan = buildManagerPlan()
    managers.value = plan.created
    managers.value.forEach((manager) => {
      // T05d：显式标注预览态 —— 面板据此把「快检心跳」显示为"未启动（预览态）"，
      //   而不是给出「未出拍」那个会被读成"通道坏了"的误导性读数。
      manager.probeMode = 'preview'
      manager.lastReason = '预览态：仅检测与建议，不会自动切换（点「启动调度」后接管出口）'
    })
    notifyUpdate()
  }

  // T05：预览态**重建**。
  // 为什么需要：`ensurePreviewManagers` 只在"一个 Manager 都没有"时才建，
  // 而「配置页保存」与「强制接管开关」之后必须**重建**才能反映新配置 ——
  // 旧版这两处都调 `ensurePreviewManagers()`，在预览态下等于什么都没做（列表还是旧的）。
  const rebuildPreviewManagers = () => {
    if (!kernelApi.running) return
    managers.value = []
    ensurePreviewManagers()
  }

  const syncManagersFromKernel = (logChanges = false, shouldNotify = false) => {
    let changed = false
    managers.value.forEach((manager) => {
      if (manager.syncExternalState(!logChanges)) changed = true
    })
    if (changed && shouldNotify) notifyUpdate()
    return changed
  }

  // ---- T02：主动同步内核出站表 ----
  // 宿主只在「代理页挂载 / 切换节点后」调 refreshProviderProxies()，**没有定时轮询**，
  // 而面板读的正是 kernelApi.proxies 这张表 ⇒ 不补这一步，用户在 App 里新加的出站
  // 不会出现在「策略组与测试网址配置」页（这正是"出站增一个，这边同步增加吗"的真实答案）。
  // 放在工厂作用域而非组件 setup 里：它是数据层行为，不依赖面板是否打开，也便于 harness 直接断言。
  const syncingOutbounds = ref(false)
  const syncOutbounds = async (silent = true) => {
    // 宿主版本差异：老宿主可能没有这个接口 ⇒ 静默降级，绝不抛错
    if (typeof kernelApi.refreshProviderProxies !== 'function') {
      if (!silent) Plugins.message.warn('当前宿主未提供出站刷新接口，切到 App 的「代理」页停留一次即可同步')
      return false
    }
    if (!kernelApi.running) {
      if (!silent) Plugins.message.info('内核未运行，无法同步出站列表')
      return false
    }
    syncingOutbounds.value = true
    try {
      await kernelApi.refreshProviderProxies()
      notifyUpdate()
      return true
    } catch (err) {
      if (silent) console.warn('[smart-policy] 出站表同步失败', err)
      else Plugins.message.error(`同步出站失败：${err.message || err}`)
      return false
    } finally {
      syncingOutbounds.value = false
    }
  }

  // T05/D3：「立即全检」的语义收敛。
  // 旧版：未启动时 `await start()` —— 按钮写着"检测"，实际副作用是**启动常驻调度** +
  //   改写持久配置 + 置 `Plugin.status = 1`。这是"文案撒谎"：用户以为只是看一眼结果。
  // 现在：只做检测。未启动就先建预览实例（不落盘、不置 isRunning），再跑一轮游离会话检测；
  //   预览态下这一轮**只出结论不切节点**（D4），结论直接显示在面板的「决策理由」里。
  const checkNow = async () => {
    if (!kernelApi.running) {
      Plugins.message.info('内核未运行，无法检测')
      return
    }
    if (!managers.value.length) ensurePreviewManagers()
    const targets = managers.value.filter((manager) => manager.enabled)
    if (!targets.length) {
      Plugins.message.warn('没有已启用调度的策略组')
      return
    }
    // 手动「全检」= 一次游离会话的单轮操作，不隶属运行代际（预览态也能用）
    await asyncPool(2, targets, (manager) => manager.checkAll(createDetachedSession()))
    notifyUpdate()
  }

  const setGroupEnabled = (group, enabled) => {
    const config = schedulerConfig.value.groups.length ? schedulerConfig.value : loadSchedulerConfig(getOptions().managedGroups)
    const groups = config.groups.map((item) => (item.name === group ? { ...item, enabled: Boolean(enabled) } : item))
    saveSchedulerConfig({ groups })
    const manager = managers.value.find((item) => item.policyGroup === group)
    if (manager) {
      manager.enabled = Boolean(enabled)
      manager.lastReason = enabled ? '已开启智能调度，等待下一轮检测' : '已关闭智能调度'
      manager.addLog(enabled ? 'success' : 'muted', manager.lastReason)
      if (enabled) manager.checkAll(createDetachedSession()).catch((error) => (lastError.value = error.message || String(error)))
    }
    notifyUpdate()
  }

  let modalSyncTimer = null
  let outboundSyncTimer = null

  // 定时器清理的唯一实现，afterClose / onUnmounted / onDispose 三处共用
  const stopModalSync = () => {
    if (modalSyncTimer) {
      clearInterval(modalSyncTimer)
      modalSyncTimer = null
    }
  }

  // T02：出站表同步定时器与上者同构 —— 必须在 afterClose 也清掉，
  // 否则关掉面板后还会一直敲内核 /proxies。
  const stopOutboundSync = () => {
    if (outboundSyncTimer) {
      clearInterval(outboundSyncTimer)
      outboundSyncTimer = null
    }
  }

  const openUI = () => {
    ensurePreviewManagers()
    injectStyle()
    const modal = Plugins.modal({
      title: Plugin.name,
      width: 1280,
      cancelText: '关闭',
      submit: false,
      afterClose: () => {
        stopModalSync()
        stopOutboundSync()
        modal.destroy()
      }
    })
    const { defineComponent, onMounted, onUnmounted } = Vue
    const component = defineComponent({
      template: `
        <div class="ai-switcher">
          <!-- Topbar -->
          <div class="ais-topbar">
            <div class="ais-brand">
              <div class="ais-brand-head">
                <span class="ais-app-title">分流策略智能调度 <span class="ais-app-ver" style="font-size:12px;font-weight:400;opacity:.65">{{ version }}</span></span>
                <span class="ais-status-chip" :class="isRunning ? 'running' : 'stopped'">
                  <i class="ais-pulse-dot"></i>
                  {{ isRunning ? '调度中' : '已停止' }}
                </span>
              </div>
              <div class="ais-brand-sub">语义自适应服务画像，非线性体感时延打分，冷启动自适应暖机与跨组防踩踏保护。</div>
            </div>
            <div class="ais-actions">

          // T02：叠加用户自定义顺序（没拖过时 groupOrder 为空，等价于纯内核顺序）
          return applyGroupOrder(detectManagedGroups())
        })
        const activeGroup = computed(() => groups.value.find((group) => group.policyGroup === activePolicyGroup.value) || groups.value[0] || null)
        const currentManager = computed(() => managers.value.find((m) => m.policyGroup === activePolicyGroup.value) || managers.value[0] || null)
        const activeDraft = computed(() => draftGroups.value.find((item) => item.name === activeDraftName.value) || draftGroups.value[0] || null)

        const summaryCards = computed(() => {
          viewTick.value
          const enabledCount = managers.value.filter((manager) => manager.enabled).length
          const g = activeGroup.value
          return [
            { label: '运行状态', value: isRunning.value ? '已启动运行' : '未启动' },
            { label: '开启策略组', value: managers.value.length ? `${enabledCount} / ${managers.value.length} 个` : `${availableGroups.value.length} 个` },
            { label: '当前大区候选池', value: g?.candidateGroup || '-' },
            {
              label: '当前选用节点',
              value: g?.current || '-',
              title: g ? `节点来源：${g.kernelNowSource}${g.kernelNowAt ? `（${formatLogTime(g.kernelNowAt)} 读回）` : ''}` : ''
            },
            // T04：快检通道的可观测面。右侧那个延迟值**只有探测会写** ⇒ 显示具体毫秒数就是「通道在跑」的硬证据。
            // T05d：读数必须带**来源**（快检 / 全池轮 / 粘性）—— 否则三个来源压成一个数字，分不清是"哪条通道测的"。
            {
              label: '当前节点快检',
              value: g ? `${g.currentCheckText} · ${g.currentDelayText}` : '-',
              title: g
                ? `快检周期 ${g.currentCheckText}；当前读数 ${g.currentDelayText}（来源：${g.currentDelaySource}）\n${g.currentStatusText}`
                : ''
            },
            // T05c 新增「快检心跳」。
            // ★ 判据：这个数字必须**持续增长**（每 currentCheckInterval 一拍）。
            //   不涨 ⇒ 通道真死了，或被上一拍卡住（旧版这两种都不可见，面板只会恒显 `-`）。
            //   在涨而「当前节点快检」恒为 `-` ⇒ 拍到了但没产出，看 title 里的「状态」。
            // T05d：标题改用 currentStatusText（**单一来源**推导）—— 旧版用 `note || '正常'`，
            //   而"探测失败但未达阈值"那条路径不写 note ⇒ 出现「延迟 - + 状态 正常」的自相矛盾。
            {
              label: '快检心跳',
              value: g ? g.currentTickText : '-',
              title: g
                ? `已执行 ${g.currentTickCount} 拍；最近一拍 ${g.lastCurrentTickAt ? formatLogTime(g.lastCurrentTickAt) : '无'}；\n状态：${g.currentStatusText}；本窗口连续快检失败 ${g.fastFailCount} 次${g.roundStaleNote ? `；\n⚠️ ${g.roundStaleNote}` : ''}`
                : '快检通道未运行'
            },
            { label: '近 1 小时切换', value: g ? `${g.switchesLastHour} / ${g.maxSwitchPerHour} 次` : '-' }
          ]
        })

        const displayRows = computed(() => {
          if (!activeGroup.value) return []
          let list = [...activeGroup.value.rows]

          if (searchQuery.value.trim()) {
            const q = searchQuery.value.trim().toLowerCase()
            list = list.filter((r) => r.id.toLowerCase().includes(q))
          }

          // T05/D7：过滤阈值必须与**决策阈值**同源（决策用 `options.minSwitchSuccessRatio`）。
          // 旧版硬编码 0.45 —— 今天恰好同值所以看不出问题，一旦阈值可调，两者立刻分叉，
          // 用户会看到「勾了'只看健康'却混进不合格节点」。阈值从 toView() 的 minRatio 来。
          const minRatioRaw = Number(activeGroup.value.minRatio)
          const minRatio = Number.isFinite(minRatioRaw) ? minRatioRaw : 0.45
          if (onlyHealthy.value) {
            list = list.filter((r) => r.stateCls === 's-normal' && r.successRatioNum >= minRatio)
          }

          list.sort((a, b) => {
            let vA, vB
            if (sortKey.value === 'score') {
              vA = a.scoreNum
              vB = b.scoreNum
            } else if (sortKey.value === 'ratio') {
              vA = a.successRatioNum
              vB = b.successRatioNum
            } else if (sortKey.value === 'ewma') {
              vA = a.ewmaNum
              vB = b.ewmaNum
            } else if (sortKey.value === 'fail') {
              vA = a.failureCount
              vB = b.failureCount
            } else if (sortKey.value === 'name') {
              return sortDesc.value ? b.id.localeCompare(a.id, 'zh-Hans-CN') : a.id.localeCompare(b.id, 'zh-Hans-CN')
            } else if (sortKey.value.startsWith('svc_')) {
              const k = sortKey.value.replace('svc_', '')
              const sA = a.services.find((s) => s.key === k)
              const sB = b.services.find((s) => s.key === k)
              vA = sA?.delay ?? 999999
              vB = sB?.delay ?? 999999
            } else {
              vA = a.scoreNum
              vB = b.scoreNum
            }
            return sortDesc.value ? vB - vA : vA - vB
          })

          return list
        })

        const handleSort = (key) => {
          if (sortKey.value === key) {
            sortDesc.value = !sortDesc.value
          } else {
            sortKey.value = key
            sortDesc.value = key === 'name' ? false : true
          }
        }

        const handleManualSwitch = async (nodeId) => {
          const manager = currentManager.value
          if (!manager) return
          const targetNode = manager.nodes.find((n) => n.id === nodeId)
          if (!targetNode) return
          if (manager.switching) return
          // 已是当前节点：整行都绑了点击，用户选个文字就会触发，这里必须静默短路，
          // 否则会走进切换流程并写入 15 分钟冷却，把自动调度冻住。
          if (manager.currentNodeName() === nodeId) return
          try {
            const applied = await manager.switchTo(targetNode, '用户手动切换')
            notifyUpdate()
            if (applied) {
              Plugins.message.success(`已切换到节点: ${nodeId}`)
            } else {
              Plugins.message.info(manager.lastReason || `切换未生效: ${nodeId}`)
            }
          } catch (err) {
            Plugins.message.error(`切换失败: ${err.message || err}`)
          }
        }

        const handleCheckCurrentGroup = async () => {
          const manager = currentManager.value
          if (!manager) return
          checkingGroup.value = true
          try {
            await manager.checkAll(createDetachedSession())
            Plugins.message.success(`${manager.policyGroup} 巡检完成`)
          } catch (err) {
            Plugins.message.error(`检测失败: ${err.message || err}`)
          } finally {
            checkingGroup.value = false
            notifyUpdate()
          }
        }

        const copyConfigGroups = (config) => config.groups.map((item) => ({ ...item }))
        const openConfigPanel = () => {
          if (showConfig.value) {
            showConfig.value = false
            return
          }
          captureConfigNotes()
          const config = loadSchedulerConfig(getOptions().managedGroups)
          draftGroups.value = orderedDraft(copyConfigGroups(config))
          activeDraftName.value = draftGroups.value[0]?.name || ''
          showConfig.value = true
        }
        const resetDraft = () => {
          const config = loadSchedulerConfig()
          draftGroups.value = orderedDraft(copyConfigGroups(config))
          activeDraftName.value = draftGroups.value[0]?.name || ''
        }
        const isDraftSelected = (name) => draftGroups.value.some((item) => item.name === name)
        const selectDraftGroup = (name) => {
          if (isDraftSelected(name)) activeDraftName.value = name
        }
        const toggleDraftGroup = (name, enabled) => {
          if (enabled) {
            if (!isDraftSelected(name)) draftGroups.value = [...draftGroups.value, defaultGroupConfig(name, true)]
            activeDraftName.value = name
          } else {
            draftGroups.value = draftGroups.value.filter((item) => item.name !== name)
            if (activeDraftName.value === name) activeDraftName.value = draftGroups.value[0]?.name || ''
          }
        }
        const profileLabelFor = (name) => matchServiceProfile(name).label
        const resetDraftServices = () => {
          if (!activeDraft.value) return
          activeDraft.value.servicesText = defaultGroupConfig(activeDraft.value.name, activeDraft.value.enabled).servicesText
        }
        const saveDraft = async () => {
          const groups = draftGroups.value.map((item) => normalizeGroupConfig(item)).filter(Boolean)
          if (!groups.length) {
            Plugins.message.error('至少选择一个策略组')
            return
          }
          // T02：保存时按用户顺序落盘 —— custom-config 的数组顺序
          // 同时也是工作台「策略组列表」与重载后的 Manager 顺序。
          saveSchedulerConfig({ groups: orderedDraft(groups) })
          showConfig.value = false
          if (isRunning.value) await start()
          else rebuildPreviewManagers()
          Plugins.message.success('自定义调度配置已保存')
        }

        // T01c「强制接管」开关：放行/收回「共享子资源·兜底出口·保留组」。
        // 之所以做成开关而不是删掉守卫：这些组被多个 Manager / 其它插件同时写入会互相打架，
        // 默认必须安全；用户显式要求时才放行，并且随时能收回来。
        const forceManagedOn = computed(() => forceManagedGroups.value.length > 0)
        const forceManagedNames = computed(() => forceManagedGroups.value.join('、') || '（无）')

        // ---- T05：自动启动开关（真实可关）----
        // 旧版四处读 `Plugin.AutoStart` 却从不定义它 ⇒ 恒真 ⇒ 界面里关不掉。
        // 现在读写的是 localStorage 载体，关掉后重载/开机都不会自动接管。
        const autoStartOn = computed(() => autoStart.value)
        const toggleAutoStart = (on) => {
          setAutoStart(on)
          Plugins.message[on ? 'success' : 'info'](
            on ? '已开启自动启动：重载插件 / 内核重启后会自动接管调度' : '已关闭自动启动：重载后需手动点「启动调度」'
          )
        }

        // ---- T05：参数自洽校正提示 ----
        // 校正本身发生在 `getOptions()` 里（快检失效窗口 < 阈值×周期 ⇒ 自动抬到刚好自洽）。
        // 这里只负责"把它显示出来" —— 静默改参数是比不校验更糟的做法（用户以为用的是自己的数）。
        const configNotes = ref([])
        const captureConfigNotes = () => {
          try {
            configNotes.value = getOptions().__adjustments || []
          } catch (err) {
            configNotes.value = []
          }
        }
        const toggleForceManaged = async (on) => {
          const next = on ? [...DEFAULT_FORCE_MANAGED] : []
          setForceManaged(next)
          Plugins.message[on ? 'success' : 'info'](
            on
              ? `已开启强制接管：${DEFAULT_FORCE_MANAGED.join('、')}`
              : '已关闭强制接管：共享组/兜底组恢复结构性剔除'
          )
          try {
            if (isRunning.value) await start()
            else rebuildPreviewManagers()
          } catch (err) {
            lastError.value = err.message || String(err)
          }
          draftGroups.value = orderedDraft(loadSchedulerConfig(getOptions().managedGroups).groups.map((item) => ({ ...item })))
          activeDraftName.value = draftGroups.value[0]?.name || ''
          notifyUpdate()
        }
        // ---- T02：拖动排序 ----
        // 两个列表共用一份 groupOrder：配置页可选组列表（scope='config'，含未接管组）、
        // 工作台策略组列表（scope='rail'，只有已接管组）。
        // 只有手柄能发起拖拽 —— 整行可拖会和「点击选中 / 勾选复选框」抢事件。
        const dragScope = ref('')
        const dragFrom = ref('')
        const dragOver = ref('')
        const dragAfter = ref(false)
        const resetDrag = () => {
          dragScope.value = ''
          dragFrom.value = ''
          dragOver.value = ''
          dragAfter.value = false
        }
        const onDragStart = (scope, name, event) => {
          dragScope.value = scope
          dragFrom.value = name
          dragOver.value = ''
          dragAfter.value = false
          try {
            event.dataTransfer.effectAllowed = 'move'
            event.dataTransfer.setData('text/plain', name)
            // 拖拽影像用整行，默认只有那个小手柄会显得很怪
            const row = event.currentTarget?.closest('.ais-drag-row')
            if (row) event.dataTransfer.setDragImage(row, 14, 14)
          } catch (err) {}
        }
        const onDragOver = (scope, name, event) => {
          // 跨列表不接（配置页的顺序不能靠拖到工作台列表上改）
          if (!dragFrom.value || dragScope.value !== scope || dragFrom.value === name) return
          event.preventDefault()
          try {
            event.dataTransfer.dropEffect = 'move'
          } catch (err) {}
          const rect = event.currentTarget.getBoundingClientRect()
          dragAfter.value = event.clientY > rect.top + rect.height / 2
          dragOver.value = name
        }
        const onDragLeave = (scope, name, event) => {
          if (dragScope.value !== scope || dragOver.value !== name) return
          // dragleave 会在行内部子元素之间反复冒泡；不放行的话指示线会闪。
          const to = event?.relatedTarget
          if (to && event.currentTarget?.contains(to)) return
          dragOver.value = ''
          dragAfter.value = false
        }
        const dropClass = (scope, name) => {
          if (!dragFrom.value || dragScope.value !== scope || dragOver.value !== name) return ''
          return dragAfter.value ? 'drop-after' : 'drop-before'
        }
        const rankOf = (seq) => {
          const rank = new Map()
          seq.forEach((name, index) => rank.set(name, index))
          return rank
        }
        // 草稿列表一律按用户顺序排好再展示/保存。
        // 为什么必须统一：展示顺序（availableGroups 读顺序表）与草稿顺序（draftGroups 数组）
        // 是两个来源，不对齐就会出现「拖完看着变了、保存下去没变」和
        // 「点重置后列表顺序与勾选顺序对不上」。顺序表为空时原样返回（= 老版本行为）。
        const orderedDraft = (groups) => {
          if (!groupOrder.value.length) return groups.slice()
          const rank = rankOf(groupOrder.value)
          const fallback = Number.MAX_SAFE_INTEGER
          return groups
            .slice()
            .sort((a, b) => (rank.has(a.name) ? rank.get(a.name) : fallback) - (rank.has(b.name) ? rank.get(b.name) : fallback))
        }
        // 配置页拖完：只改「草稿顺序」（本页本来就是草稿语义，点「保存并应用」才生效）。
        // 已接管组之间的先后决定保存后的 Manager 顺序；未接管组只影响展示位置。
        const dropConfigGroup = (from, to, after) => {
          saveGroupOrder(reorderInList(availableGroups.value, from, to, after))
          // 展示顺序已经由 availableGroups 自己重算；这里同步草稿顺序，
          // 免得「列表看起来是新的、保存下去却是旧的」。
          draftGroups.value = orderedDraft(draftGroups.value)
          notifyUpdate()
        }
        // 工作台拖完：立即落盘（这里没有草稿态）。顺序同时写进 custom-config，
        // 这样「保存并应用」和「重载插件」后顺序都不会丢。
        const persistManagedOrder = (seq) => {
          const config = schedulerConfig.value.groups.length ? schedulerConfig.value : loadSchedulerConfig(getOptions().managedGroups)
          const byName = new Map(config.groups.map((item) => [item.name, item]))
          const ordered = seq.map((name) => byName.get(name)).filter(Boolean)
          if (!ordered.length) return
          config.groups.forEach((item) => {
            if (!ordered.some((group) => group.name === item.name)) ordered.push(item)
          })
          saveSchedulerConfig({ groups: ordered })
          const managerByName = new Map(managers.value.map((manager) => [manager.policyGroup, manager]))
          const reordered = ordered.map((item) => managerByName.get(item.name)).filter(Boolean)
          // 长度不等说明有组被结构性剔除了，这时不重排 managers，避免丢卡片
          if (reordered.length === managers.value.length) managers.value = reordered
        }
        const dropRailGroup = (from, to, after) => {
          const managed = groups.value.map((group) => group.policyGroup)
          const seq = reorderInList(managed, from, to, after)
          // 把已接管组的新序列「写回它们在总顺序里原本占用的那些槽位」，
          // 未接管组不会因为一次拖动被整体挤走。
          const inManaged = new Set(managed)
          const cursor = seq[Symbol.iterator]()
          const merged = []
          groupOrder.value.forEach((name) => {
            if (!inManaged.has(name)) {
              merged.push(name)
              return
            }
            const next = cursor.next()
            merged.push(next.done ? name : next.value)
          })
          let rest = cursor.next()
          while (!rest.done) {
            merged.push(rest.value)
            rest = cursor.next()
          }
          saveGroupOrder(merged)
          persistManagedOrder(seq)
          notifyUpdate()
        }
        const onDrop = (scope, name) => {
          const from = dragFrom.value
          const after = dragAfter.value
          resetDrag()
          if (!from || from === name) return
          if (scope === 'config') dropConfigGroup(from, name, after)
          else if (scope === 'rail') dropRailGroup(from, name, after)
        }

        const healthClass = (group) => {
          const ratio = Number(String(group.bestRatio).replace('%', ''))
          if (ratio >= 80) return 'good'
          if (ratio >= 45) return 'warn'
          return 'poor'
        }
        onMounted(() => {
          captureConfigNotes()
          syncManagersFromKernel(false, true)
          // 打开面板先拉一次出站表：否则要看的是「上次宿主刷新时的快照」
          syncOutbounds(true)
          stopModalSync()
          modalSyncTimer = setInterval(() => syncManagersFromKernel(true, true), MODAL_SYNC_INTERVAL)
          stopOutboundSync()
          outboundSyncTimer = setInterval(() => syncOutbounds(true), OUTBOUND_SYNC_INTERVAL)
        })
        // 定时器统一走 stop*：afterClose 是主路径，此处仅兜组件异常卸载
        onUnmounted(() => {
          stopModalSync()
          stopOutboundSync()
        })

        return {
          version: VERSION,
          isRunning,
          lastError,
          checking,
          checkingGroup,
          showConfig,
          showLogs,
          searchQuery,
          onlyHealthy,
          sortKey,
          sortDesc,
          groups,
          availableGroups,
          activeGroup,
          activeDraft,
          summaryCards,
          displayRows,
          handleSort,
          handleManualSwitch,
          handleCheckCurrentGroup,
          setActiveGroup: (policyGroup) => (activePolicyGroup.value = policyGroup),
          openConfigPanel,
          resetDraft,
          isDraftSelected,
          selectDraftGroup,
          toggleDraftGroup,
          profileLabelFor,
          resetDraftServices,
          saveDraft,
          forceManagedOn,
          forceManagedNames,
          toggleForceManaged,
          autoStartOn,
          toggleAutoStart,
          configNotes,
          dragFrom,
          dropClass,
          resetDrag,
          onDragStart,
          onDragOver,
          onDragLeave,
          onDrop,
          syncingOutbounds,
          syncOutbounds,
          toggleGroup: setGroupEnabled,
          healthClass,
          stop,
          async handleStart() {
            try {
              await start()
            } catch (error) {
              lastError.value = error.message || String(error)
              Plugins.message.error(lastError.value)
            }
          },
          async handleCheck() {
            checking.value = true
            try {
              await checkNow()
            } catch (error) {
              lastError.value = error.message || String(error)
              Plugins.message.error(lastError.value)
            } finally {
              checking.value = false
            }
          }
        }
      }
    })
    modal.setContent(component)
    modal.open()
  }

  return {
    onRun: () => openUI(),
    onReady: () => {
      if (autoStart.value) {
        setTimeout(() => start().catch((error) => (lastError.value = error.message || String(error))), 3000)
      }
    },
    // T05/D8：不再 `stop(false, true); return start(config)`。
    // `start()` 内部现在是「先构建、成功后才停旧」，所以配置有问题时**旧 runtime 不会被误杀**
    // （对齐 B 的「先 build 再 stop」）。这里再 stop 一次反而会把这条保护抵消掉。
    onConfigure: async (config) => start(config),
    // ---- T03/D10（部分修）：与 onDispose 对称的恢复入口 ----
    // 原实现只有 onDispose、没有 onEnabled —— 于是「App 内禁用 → 再启用」之后调度不会自己回来。
    // 这三个钩子（onEnabled / onDisabled / onConfigure）**不经 triggers 门控**，写了就生效。
    onEnabled: () => {
      if (autoStart.value) {
        setTimeout(() => start().catch((error) => (lastError.value = error.message || String(error))), 1500)
      }
    },
    onDisabled: () => stop(true, true),
    // ⚠️ onReload **需要**在 App 内把 `on::reload` 勾进触发器才会被调用（未勾选 = 永不执行）。
    // 勾上它，「重载插件」这条路径才具备自恢复能力（D10 的另一半）。T05 已把它并入
    // `apply-triggers.cjs` 的必需清单，脚本一条命令即可补齐。
    onReload: () => {
      if (autoStart.value) {
        setTimeout(() => start().catch((error) => (lastError.value = error.message || String(error))), 1500)
      }
    },
    onCoreStarted: () => {
      if (autoStart.value) {
        setTimeout(() => start().catch((error) => (lastError.value = error.message || String(error))), 1500)
      }
    },
    onCoreStopped: () => stop(true, true),
    onDispose: () => {
      stopModalSync()
      return stop(false, true)
    }
  }
